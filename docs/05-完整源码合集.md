## 4. `app\build.gradle`

**说明**：应用模块构建脚本（compileSdk 10 / compileOnly 私有桩）

```gradle
/*
 * PMCA_IntervalCapture 机内间隔拍照应用 —— app 模块构建脚本
 *
 * 关键点（PMCA 平台硬性约束）：
 *   1. 相机运行时是 Android 2.3.7（API Level 10），因此 compileSdkVersion = 10、minSdkVersion = 10；
 *   2. 索尼私有 API（com.sony.scalar.*）只存在于相机固件中，不能在运行时打进 APK，
 *      所以必须用 provided / compileOnly 方式引入（只参与编译，不参与打包）；
 *   3. 目标 JAVA 版本为 1.7（API 10 时代的 DEX 格式最稳妥），不使用 lambda、Stream 等新语法；
 *   4. 不申请任何视频录制相关能力，本应用只调用静态照片捕获接口。
 */
apply plugin: 'com.android.application'

android {
    compileSdkVersion 10
    buildToolsVersion "30.0.3"

    defaultConfig {
        // 包名固定，PMCA-RE 刷入后以该包名出现在相机的“应用程序”列表中
        applicationId "com.pmca.intervalcapture"
        minSdkVersion 10
        targetSdkVersion 10
        versionCode 2
        versionName "1.0.1"
        // 相机屏幕为 640x480，不支持多语言资源目录以外的限定符，这里只保留默认配置
    }

    compileOptions {
        // 必须使用 1.7 及以下语法，Android 2.3 的 ART/Dalvik 与索尼运行时对高版本字节码支持有限
        sourceCompatibility JavaVersion.VERSION_1_7
        targetCompatibility JavaVersion.VERSION_1_7
    }

    // 关闭 lint 的致命错误中断，避免旧 SDK 的检查项阻断打包
    lintOptions {
        abortOnError false
        checkReleaseBuilds false
    }

    // 不使用任何原生库与渲染脚本，减小编排出错面
    packagingOptions {
        exclude 'META-INF/LICENSE.txt'
        exclude 'META-INF/NOTICE.txt'
    }
}

dependencies {
    /*
     * 编译期依赖（只参与编译，不会打进 APK）：
     *
     *   libs/sony-pmca-stubs.jar
     *     Sony PMCA 私有 API 的编译期桩（com.sony.scalar.hardware.CameraEx、
     *     com.sony.scalar.sysutil.* 等）。这些类的真正实现在相机固件里，
     *     所以只能用 compileOnly / provided 引入，绝不能打包进 APK，
     *     否则会和相机固件里的同名类冲突，导致应用一启动就崩溃。
     *
     *   libs/android-4.1.1.4.jar
     *     Android 4.1.1（API 16）的 AOSP 类库存根，仅作为编译期对照；
     *     相机实际运行时是 Android 2.3.7（API 10），因此代码里只能调用
     *     API 10 就已存在的 android.* 接口（本工程已严格检查过）。
     *
     * 若使用索尼官方 Camera Apps SDK，把官方 jar 放进 app/libs 即可，
     * 下面的 fileTree 会自动把它一起纳入编译。
     */
    compileOnly fileTree(dir: 'libs', include: ['*.jar'])
}
``````

```xml
<?xml version="1.0" encoding="utf-8"?>
<!--
  PMCA_IntervalCapture 机内间隔拍照应用 —— 应用清单

  平台事实（务必理解，否则会误以为是普通 Android APK）：
    1. 相机运行时是 Android 2.3.7（API Level 10），没有普通意义上的 Home 桌面；交互只用机身实体按键（本应用不注册任何触摸监听）；
    2. 相机固件通过 com.android.server.DAConnectionManagerService.AppInfoReceive 广播
       得知“当前哪个 Activity 在前台”，因此本应用启动时会发送该广播（见 BaseActivity 逻辑）；
    3. 应用退出后由固件重新启动原生拍照界面，相当于“返回相机原生 UI”；
    4. 只申请静态照片拍摄与 SD 卡写入相关权限，不申请任何音视频/联网权限。
-->
<manifest xmlns:android="http://schemas.android.com/apk/res/android"
    package="com.pmca.intervalcapture"
    android:versionCode="2"
    android:versionName="1.0.1">

    <!-- 允许打开相机硬件（静态照片捕获必需） -->
    <uses-permission android:name="android.permission.CAMERA" />
    <!-- 允许写入相机 SD 卡（日志文件、参数配置写入） -->
    <uses-permission android:name="android.permission.WRITE_EXTERNAL_STORAGE" />
    <!-- 允许在长时间拍摄期间保持 CPU 唤醒，避免息屏导致间隔计时漂移 -->
    <uses-permission android:name="android.permission.WAKE_LOCK" />

    <!-- 声明使用相机硬件；required=false 便于在部分固件上仍能启动并给出中文提示 -->
    <uses-feature android:name="android.hardware.camera" android:required="false" />

    <!--
      必须显式声明 uses-sdk：
        1. 相机运行时是 Android 2.3.7（API Level 10），minSdkVersion 写 10 才与目标平台一致；
        2. PMCA-RE 安装器（pmca/apk/__init__.py 的 getMinSdkVersion()）会读取这个元素，
           缺失时它会抛异常并打印 "Warning: Invalid apk file"，部分固件的应用安装器
           同样依赖该字段判断兼容性，缺失可能导致安装被拒。
    -->
    <uses-sdk android:minSdkVersion="10" android:targetSdkVersion="10" />

    <application
        android:label="@string/app_name"
        android:icon="@drawable/ic_launcher"
        android:allowBackup="false"
        android:theme="@style/AppTheme">

        <!--
          唯一的 Activity：参数设置页、运行状态页、完成/错误弹窗都在这一个 Activity 内切换，
          这样相机固件始终认为前台只有一个界面，减少切换时的画面闪烁与预览中断。
        -->
        <activity
            android:name=".CaptureActivity"
            android:label="@string/app_title"
            android:screenOrientation="landscape"
            android:configChanges="orientation|keyboardHidden"
            android:launchMode="singleInstance"
            android:windowSoftInputMode="stateAlwaysHidden">

            <!-- 作为应用入口，出现在相机“应用程序”列表中 -->
            <intent-filter>
                <action android:name="android.intent.action.MAIN" />
                <category android:name="android.intent.category.LAUNCHER" />
            </intent-filter>
        </activity>

        <!--
          接收相机固件在“应用启动完成 / 应用退出完成”时发出的广播。
          固件用这两个广播维护应用生命周期，某些机型若缺少接收器会出现
          “退出后黑屏”或“再次进入应用卡住”的问题。
        -->
        <receiver android:name=".SystemEventReceiver">
            <intent-filter>
                <action android:name="com.android.server.DAConnectionManagerService.BootCompleted" />
                <action android:name="com.android.server.DAConnectionManagerService.ExitCompleted" />
                <category android:name="ANDROID.INTENT.CATEGORY.DEFAULT" />
            </intent-filter>
        </receiver>
    </application>
</manifest>
``````

```java
        // ③ 向相机固件申报“当前前台应用”
        notifyAppInfo();

        // ④ 申请 CPU 唤醒锁（只在拍摄任务期间持有）
        acquireWakeLock();

        // ⑤ 相机在进入本界面时打开，退出时释放
        openCameraAsync();

        // ⑥ 启动界面刷新循环
        uiHandler.postDelayed(tickRunnable, TICK_MS);
    }

    /**
     * 构建 View 树：
     *   FrameLayout
     *     ├─ SurfaceView（预览）
     *     └─ CanvasView（覆盖层，透明背景，只在信息区绘制）
     */
    private void buildViewTree() {
        FrameLayout root = new FrameLayout(this);
        root.setBackgroundColor(0xFF000000);

        previewView = new SurfaceView(this);
        previewView.getHolder().setType(SurfaceHolder.SURFACE_TYPE_PUSH_BUFFERS);
        previewView.getHolder().addCallback(new SurfaceHolder.Callback() {
            @Override
            public void surfaceCreated(SurfaceHolder holder) {
                // 预览显示面就绪：如果相机已打开则绑定显示面
                if (captureService != null && captureService.isOpened() && settings.isPreviewEnabled()) {
                    captureService.startPreview(holder);
                    previewVisible = true;
                }
            }

            @Override
            public void surfaceChanged(SurfaceHolder holder, int format, int width, int height) {
            }

            @Override
            public void surfaceDestroyed(SurfaceHolder holder) {
                previewVisible = false;
                if (captureService != null) {
                    captureService.stopPreview();
                }
            }
        });
        FrameLayout.LayoutParams pvLp = new FrameLayout.LayoutParams(
                RunScreen.PREVIEW_RECT.width(), RunScreen.PREVIEW_RECT.height());
        pvLp.leftMargin = RunScreen.PREVIEW_RECT.left;
        pvLp.topMargin = RunScreen.PREVIEW_RECT.top;
        previewView.setVisibility(View.GONE);     // 参数页不需要预览
        root.addView(previewView, pvLp);

        canvasView = new CanvasView(this);
        root.addView(canvasView, new FrameLayout.LayoutParams(
                ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.FILL_PARENT));

        setContentView(root);
    }

    @Override
    protected void onResume() {
        super.onResume();
        AppLog.info("onResume");
        notifyAppInfo();
        uiHandler.removeCallbacks(tickRunnable);
        uiHandler.postDelayed(tickRunnable, TICK_MS);
    }

    @Override
    protected void onPause() {
        super.onPause();
        AppLog.info("onPause");
        // 被相机固件切走（例如打开播放界面）时，必须先停止拍摄并释放相机，
        // 否则固件会认为相机被占用，出现“无法拍照”。
        stopTaskInternal("应用被切换到后台，拍摄已停止", true);
        uiHandler.removeCallbacks(tickRunnable);
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        AppLog.info("onDestroy");
        uiHandler.removeCallbacksAndMessages(null);
        closeCamera();
        healthMonitor.close();
        releaseWakeLock();
    }

    /* ============================ 相机打开 / 关闭 ============================ */

    /**
     * 异步打开相机：相机打开可能需要几百毫秒，不能阻塞 UI 线程，
     * 否则界面会白屏、按键无响应。
     */
    private void openCameraAsync() {
        cameraStatusLine = "正在打开相机…";
        invalidate();
        new Thread(new Runnable() {
            @Override
            public void run() {
                final boolean ok = doOpenCamera();
                uiHandler.post(new Runnable() {
                    @Override
                    public void run() {
                        cameraOk = ok;
                        cameraStatusLine = ok ? null : "相机初始化失败：请确认机内拍摄模式为静态照片(P/A/S/M)，然后返回重试";
                        if (!ok) {
                            AppLog.error("相机打开失败");
                        }
                        invalidate();
                    }
                });
            }
        }, "open-camera").start();
    }

    /** 在后台线程里真正调用 CameraEx.open（见 CaptureService） */
    private boolean doOpenCamera() {
        if (captureService == null) {
            captureService = new CaptureService(captureCallback);
        }
        SurfaceHolder holder = null;
        if (settings.isPreviewEnabled() && previewView != null && previewView.getHolder() != null) {
            holder = previewView.getHolder();
        }
        return captureService.open(holder, settings.isLockAeAf());
    }

    private void closeCamera() {
        if (captureService != null) {
            captureService.close();
            captureService = null;
        }
        cameraOk = false;
    }

    /* ============================ 采集回调（固件线程） ============================ */

    private final CaptureCallback captureCallback = new CaptureCallback() {
        @Override
        public void onCameraOpened() {
        }

        @Override
        public void onShotTaken() {
            // 快门动作：仅用于日志，不在固件线程里刷新界面
            AppLog.info("快门已动作");
        }

        @Override
        public void onImageStored(final String fileName) {
            // 写卡完成：切回 UI 线程更新调度器计数
            uiHandler.post(new Runnable() {
                @Override
                public void run() {
                    healthMonitor.onStoreSuccess();
                    if (scheduler != null) {
                        scheduler.onShotSaved(fileName);
                    }
                    message = "已保存：" + fileName;
                    messageIsError = false;
                    invalidate();
                }
            });
        }

        @Override
        public void onCaptureFailed(final String reason) {
            uiHandler.post(new Runnable() {
                @Override
                public void run() {
                    handleCaptureFailure(reason);
                }
            });
        }

        @Override
        public void onCameraError(final int errorCode) {
            uiHandler.post(new Runnable() {
                @Override
                public void run() {
                    stopWithError("相机错误（代码 " + errorCode + "）",
                            "相机固件报告错误，拍摄已停止。请检查镜头是否安装到位、存储卡是否正常。",
                            "已保存的照片不受影响");
                }
            });
        }

        @Override
        public void onCameraClosed() {
        }
    };

    /* ============================ 主循环：刷新 + 节拍 ============================ */

    private final Runnable tickRunnable = new Runnable() {
        @Override
        public void run() {
            doTick();
            invalidate();
            uiHandler.postDelayed(this, TICK_MS);
        }
    };

    /**
     * 每 100ms 执行一次：
     *   1. 运行模式下判断是否到点，到点则调用相机拍照；
     *   2. 检查单张拍摄是否超时；
     *   3. 检查是否因为过热/SD 卡问题需要中止。
     */
    private void doTick() {
        if (mode != MODE_RUNNING || scheduler == null) {
            return;
        }
        int state = scheduler.getState();
        if (state != IntervalScheduler.STATE_RUNNING) {
            return;
        }
        // 单张拍摄超时保护
        if (scheduler.isCaptureTimeout()) {
            AppLog.error("单张拍摄超时");
            if (captureService != null) {
                captureService.cancelCapture();
            }
            stopWithError("拍摄超时", "相机在 30 秒内没有完成写入，可能是存储卡写入过慢或机身异常。",
                    "已拍照片均已保存");
            return;
        }
        // 是否到点
        if (scheduler.shouldTriggerNow()) {
            triggerShot();
        }
    }

    /**
     * 触发一张拍摄：
     *   1. 先做健康巡检（SD 卡 / 空间 / 温度），有问题立即停止；
     *   2. 调用 CaptureService.takePicture（内部走 CameraEx → Camera.takePicture）。
     */
    private void triggerShot() {
        if (captureService == null || !captureService.isOpened()) {
            stopWithError("相机未就绪", "相机连接已断开，拍摄已停止。", "已拍照片均已保存");
            return;
        }
        // 健康巡检
        boolean firstShot = scheduler.getTakenCount() == 0 && scheduler.getFailedCount() == 0;
        if (!healthMonitor.checkBeforeShot(firstShot)) {
            stopWithError("拍摄已停止", healthMonitor.getFatalReason(), "已拍照片均已保存");
            return;
        }
        String warn = healthMonitor.getWarningText();
        if (warn != null) {
            message = warn;
            messageIsError = false;
        }
        // 标记开始：必须在调用 takePicture 之前标记，保证回调不会漏统计
        scheduler.onShotStarted();
        boolean submitted = captureService.takePicture();
        if (!submitted) {
            // 提交失败（例如上一张还没写完卡）：本张按失败处理，等下一次节拍
            scheduler.onShotFailed("拍摄请求未被接受");
        }
    }

    /* ============================ 任务控制 ============================ */

    /** 从参数页开始一轮拍摄任务 */
    private void startTask() {
        if (!cameraOk || captureService == null || !captureService.isOpened()) {
            // 相机没打开成功：重新尝试一次
            AppLog.warn("相机未就绪，重新打开");
            openCameraAsync();
            message = "相机未就绪，正在重新初始化…";
            messageIsError = true;
            invalidate();
            return;
        }
        // 保存参数，防止断电后丢失
        store.save(settings);

        scheduler = new IntervalScheduler(settings.copy());
        scheduler.setListener(schedulerListener);

        // 打开温度监控
        healthMonitor.open();

        // 按参数决定是否显示预览
        applyPreviewVisibility(settings.isPreviewEnabled());

        mode = MODE_RUNNING;
        message = "开始拍摄（" + settings.formatInterval() + " × "
                + (settings.isInfiniteCount() ? "无限" : settings.getEffectiveShotCount() + "张") + "）";
        messageIsError = false;
        scheduler.start();
        acquireWakeLock();

        // 相机硬件限制提示：间隔太短时立刻提醒
        String hw = settings.getHardwareWarning();
        if (hw != null) {
            message = hw + "（将尽力拍摄）";
        }
        invalidate();
    }

    /** 停止任务并回到参数页 */
    private void stopTaskInternal(String reason, boolean backToSettings) {
        if (scheduler != null && mode == MODE_RUNNING) {
            AppLog.info("停止拍摄：" + reason);
            if (captureService != null) {
                captureService.cancelCapture();
            }
            scheduler.stop(reason);
        }
        scheduler = null;
        healthMonitor.close();
        applyPreviewVisibility(false);
        mode = MODE_SETTINGS;
        if (backToSettings) {
            message = reason;
            messageIsError = false;
        }
        invalidate();
    }

    /** 出现不可恢复错误：停止拍摄、弹中文错误框 */
    private void stopWithError(String title, String reason, String extra) {
        AppLog.error("拍摄异常停止：" + title + " - " + reason);
        if (scheduler != null) {
            scheduler.fail(reason);
        }
        int taken = scheduler == null ? 0 : scheduler.getTakenCount();
        if (captureService != null) {
            captureService.cancelCapture();
        }
        scheduler = null;
        healthMonitor.close();
        applyPreviewVisibility(false);
        errorTitle = title;
        errorReason = reason;
        errorExtra = (taken > 0 ? "本次已成功保存 " + taken + " 张照片。" : "") + (extra == null ? "" : extra);
        mode = MODE_ERROR_DIALOG;
        invalidate();
    }

    /** 单张拍摄失败（不代表整轮任务失败）：累计失败次数过多才停止 */
    private void handleCaptureFailure(String reason) {
        AppLog.warn("单张拍摄失败：" + reason);
        boolean tooMany = healthMonitor.onStoreFailure();
        if (scheduler != null) {
            scheduler.onShotFailed(reason);
        }
        message = reason;
        messageIsError = true;
        if (tooMany) {
            stopWithError("相机异常", reason, null);
        }
        invalidate();
    }

    /* ============================ 调度器监听 ============================ */

    private final IntervalScheduler.Listener schedulerListener = new IntervalScheduler.Listener() {
        @Override
        public void onStateChanged(final int newState) {
            uiHandler.post(new Runnable() {
                @Override
                public void run() {
                    AppLog.info("状态 -> " + IntervalScheduler.stateName(newState));
                }
            });
        }

        @Override
        public void onShotSaved(final int takenCount, final String fileName) {
            // onImageStored 已经做计数处理，这里不重复计数
        }

        @Override
        public void onShotFailed(final String reason) {
        }

        @Override
        public void onFinished(final int takenCount) {
            uiHandler.post(new Runnable() {
                @Override
                public void run() {
                    showFinishDialog(takenCount);
                }
            });
        }
    };

    /** 拍摄完成：弹出“是否生成延时视频”提示，默认“否” */
    private void showFinishDialog(int takenCount) {
        int failed = scheduler == null ? 0 : scheduler.getFailedCount();
        store.addTotalTaken(takenCount);
        AppLog.info("任务完成：成功 " + takenCount + " 张，失败 " + failed + " 张");
        scheduler = null;
        healthMonitor.close();
        applyPreviewVisibility(false);
        chooseVideo = false;          // ★ 默认选中“否”，不生成视频
        finishTakenCount = takenCount;
        finishFailedCount = failed;
        mode = MODE_FINISH_DIALOG;
        invalidate();
    }

    private int finishTakenCount = 0;
    private int finishFailedCount = 0;

    /* ============================ 按键处理 ============================ */

    @Override
    public boolean onKeyDown(int keyCode, KeyEvent event) {
        int action = CameraKeyRouter.routeKeyDown(event);
        if (action == CameraKeyRouter.ACTION_NONE) {
            return super.onKeyDown(keyCode, event);
        }
        handleAction(action);
        return true;
    }

    @Override
    public boolean onKeyUp(int keyCode, KeyEvent event) {
        int action = CameraKeyRouter.routeKeyUp(event);
        if (action == CameraKeyRouter.ACTION_NONE) {
            return super.onKeyUp(keyCode, event);
        }
        return true;
    }

    /** 按键动作分发（按当前界面模式） */
    private void handleAction(int action) {
        switch (mode) {
            case MODE_SETTINGS:
                handleSettingsAction(action);
                break;
            case MODE_RUNNING:
                handleRunningAction(action);
                break;
            case MODE_FINISH_DIALOG:
                handleFinishDialogAction(action);
                break;
            case MODE_ERROR_DIALOG:
                handleErrorDialogAction(action);
                break;
            case MODE_CONFIRM_DIALOG:
                handleConfirmDialogAction(action);
                break;
            default:
                break;
        }
        invalidate();
    }

    /* -------- 参数页按键 -------- */
    private void handleSettingsAction(int action) {
        switch (action) {
            case CameraKeyRouter.ACTION_UP:
                settingsScreen.moveFocus(-1);
                break;
            case CameraKeyRouter.ACTION_DOWN:
                settingsScreen.moveFocus(1);
                break;
            case CameraKeyRouter.ACTION_LEFT:
                settings.adjust(settingsScreen.getFocusIndex(), -1);
                store.save(settings);
                break;
            case CameraKeyRouter.ACTION_RIGHT:
                settings.adjust(settingsScreen.getFocusIndex(), 1);
                store.save(settings);
                break;
            case CameraKeyRouter.ACTION_FN:
                settingsScreen.moveFocus(1);
                break;
            case CameraKeyRouter.ACTION_ENTER:
            case CameraKeyRouter.ACTION_SHUTTER:
                startTask();
                break;
            case CameraKeyRouter.ACTION_PLAY:
                showConfirmDialog("拍摄统计",
                        "成功保存过的累计张数：" + store.getTotalTaken() + " 张\n"
                                + "日志文件：SD 卡 PMCACAP/LOG.TXT\n"
                                + "参数备份：SD 卡 PMCACAP/SETTINGS.TXT",
                        "知道了", "取消", 0, 0);
                break;
            case CameraKeyRouter.ACTION_BACK:
                showConfirmDialog("退出应用？", "按“确认”返回相机原生拍摄界面。",
                        "取消", "退出", 0, CONFIRM_EXIT_APP);
                break;
            default:
                break;
        }
    }

    /* -------- 运行页按键 -------- */
    private void handleRunningAction(int action) {
        if (scheduler == null) {
            mode = MODE_SETTINGS;
            return;
        }
        int state = scheduler.getState();
        switch (action) {
            case CameraKeyRouter.ACTION_ENTER:
            case CameraKeyRouter.ACTION_SHUTTER:
                // 确认键 / 快门键：开始 ⇄ 暂停
                if (state == IntervalScheduler.STATE_RUNNING) {
                    scheduler.pause();
                    if (captureService != null) {
                        captureService.cancelCapture();
                    }
                    message = "已暂停（按确认继续，计数不重置）";
                } else if (state == IntervalScheduler.STATE_PAUSED) {
                    scheduler.resume();
                    message = "继续拍摄";
                }
                messageIsError = false;
                break;
            case CameraKeyRouter.ACTION_DOWN:
                // 下方向键：停止拍摄，回到参数设置页
                stopTaskInternal("已手动停止拍摄", true);
                break;
            case CameraKeyRouter.ACTION_UP:
                // 上方向键：查看/切换预览（运行中切换预览有热量与稳定性取舍）
                togglePreviewDuringRun();
                break;
            case CameraKeyRouter.ACTION_LEFT:
            case CameraKeyRouter.ACTION_RIGHT:
                // 运行中不允许改参数：提示用户先停止，避免节拍混乱
                message = "运行中不可修改参数，请按“下”停止后再调整";
                messageIsError = false;
                break;
            case CameraKeyRouter.ACTION_BACK:
                stopTaskInternal("已停止并退出拍摄", true);
                break;
            default:
                break;
        }
    }

    /** 运行中切换预览开关（只影响显示，不影响拍摄） */
    private void togglePreviewDuringRun() {
        boolean enable = !settings.isPreviewEnabled();
        settings.setPreviewEnabled(enable);
        applyPreviewVisibility(enable);
        message = enable ? "预览已开启（注意发热）" : "预览已关闭（降低发热）";
        messageIsError = false;
    }

    /* -------- 完成弹窗按键 -------- */
    private void handleFinishDialogAction(int action) {
        switch (action) {
            case CameraKeyRouter.ACTION_LEFT:
            case CameraKeyRouter.ACTION_RIGHT:
            case CameraKeyRouter.ACTION_UP:
            case CameraKeyRouter.ACTION_DOWN:
                chooseVideo = !chooseVideo;
                break;
            case CameraKeyRouter.ACTION_ENTER:
            case CameraKeyRouter.ACTION_SHUTTER:
                if (chooseVideo) {
                    // ★ 用户明确选择“是”后才进入视频合成流程（可选附加功能）
                    startOptionalTimelapseVideo();
                }
                backToSettingsAfterDialog();
                break;
            case CameraKeyRouter.ACTION_BACK:
                // 菜单键 = 直接选“否”
                chooseVideo = false;
                backToSettingsAfterDialog();
                break;
            default:
                break;
        }
    }

    /* -------- 错误弹窗按键 -------- */
    private void handleErrorDialogAction(int action) {
        if (action == CameraKeyRouter.ACTION_ENTER
                || action == CameraKeyRouter.ACTION_BACK
                || action == CameraKeyRouter.ACTION_SHUTTER) {
            mode = MODE_SETTINGS;
            message = errorReason;
            messageIsError = true;
        }
    }

    /* -------- 确认弹窗按键 -------- */
    private void handleConfirmDialogAction(int action) {
        switch (action) {
            case CameraKeyRouter.ACTION_LEFT:
            case CameraKeyRouter.ACTION_RIGHT:
            case CameraKeyRouter.ACTION_UP:
            case CameraKeyRouter.ACTION_DOWN:
                confirmEmphasis = confirmEmphasis == 0 ? 1 : 0;
                break;
            case CameraKeyRouter.ACTION_ENTER:
            case CameraKeyRouter.ACTION_SHUTTER:
                mode = MODE_SETTINGS;
                if (confirmPendingAction == CONFIRM_EXIT_APP && confirmEmphasis == 1) {
                    exitToCameraUi();
                }
                confirmPendingAction = 0;
                break;
            case CameraKeyRouter.ACTION_BACK:
                mode = MODE_SETTINGS;
                confirmPendingAction = 0;
                break;
            default:
                break;
        }
    }

    private void showConfirmDialog(String title, String msg, String opt0, String opt1,
                                   int emphasis, int pendingAction) {
        confirmTitle = title;
        confirmMessage = msg;
        confirmOption0 = opt0;
        confirmOption1 = opt1;
        confirmEmphasis = emphasis;
        confirmPendingAction = pendingAction;
        mode = MODE_CONFIRM_DIALOG;
        invalidate();
    }

    private void backToSettingsAfterDialog() {
        mode = MODE_SETTINGS;
        message = "已完成本次拍摄，可继续设置新的参数";
        messageIsError = false;
        invalidate();
    }

    /* ============================ 可选：延时视频合成 ============================ */

    /**
     * 【可选附加功能】把本次拍摄的照片合成为延时视频。
     * <p>
     * 设计原则（严格遵守需求）：
     *   - 照片保存永远是第一优先级，本方法只会在用户主动选择“是”之后被调用；
     *   - 默认路径完全不进入本方法；
     *   - 合成失败绝不影响、也不会删除已保存的原图。
     * <p>
     * 关于实现：
     *   PMCA SDK 没有提供“把多张照片编码成一个视频文件”的公开接口。
     *   camera firmware 自带的视频录制接口（MediaRecorder 等）只能录制实时画面，
     *   无法把 SD 卡上已有的照片编码成视频。因此本方法只做两件事：
     *     1) 在 SD 卡上生成一份“视频合成任务清单” TIMELAPSE_JOB.TXT，
     *        记录照片目录、帧率建议等信息，供电脑端工具离线合成；
     *     2) 在界面上给出明确中文提示。
     *   这样既满足“可选附加功能”的需求，又保证机内不会悄悄生成视频，
     *   也不会因为编码占用机身算力而导致过热或拍摄中断。
     */
    private void startOptionalTimelapseVideo() {
        AppLog.info("用户选择生成延时视频（可选附加功能）");
        boolean ok = TimelapseJobWriter.writeJob(finishTakenCount, settings);
        String msg;
        if (ok) {
            msg = "已生成视频合成任务清单（SD 卡 PMCACAP/TIMELAPSE_JOB.TXT），可用电脑端工具合成";
        } else {
            msg = "视频合成清单写入失败，但照片已全部保存，不受影响";
        }
        message = msg;
        messageIsError = false;
        android.widget.Toast.makeText(this, msg, android.widget.Toast.LENGTH_LONG).show();
    }

    /* ============================ 预览显隐 ============================ */

    /**
     * 控制预览视图可见性：
     *   - 打开：显示 SurfaceView 并启动预览；
     *   - 关闭：停止预览并把 SurfaceView 设为 GONE（不再占用显示带宽与传感器预览）。
     */
    private void applyPreviewVisibility(boolean visible) {
        if (previewView == null) {
            return;
        }
        if (visible) {
            previewView.setVisibility(View.VISIBLE);
            if (captureService != null && captureService.isOpened()) {
                captureService.startPreview(previewView.getHolder());
                previewVisible = true;
            }
        } else {
            if (captureService != null) {
                captureService.stopPreview();
            }
            previewVisible = false;
            previewView.setVisibility(View.GONE);
        }
        invalidate();
    }

    /* ============================ 绘制 ============================ */

    /** 触发一次重绘 */
    private void invalidate() {
        if (canvasView != null) {
            canvasView.postInvalidate();
        }
    }

    /**
     * 自绘 Canvas 视图：把三个界面的绘制委派给对应 Screen 类。
     * 背景透明，只有左侧信息区有内容，右侧留给相机预览。
     */
    private final class CanvasView extends View {

        CanvasView(android.content.Context context) {
            super(context);
            setFocusable(true);
            setFocusableInTouchMode(true);
        }

        @Override
        protected void onDraw(Canvas canvas) {
            super.onDraw(canvas);
            // 画布固定 640x480；若固件分辨率不同，做等比缩放
            int w = getWidth();
            int h = getHeight();
            if (w <= 0 || h <= 0) {
                return;
            }
            if (w != UiTheme.SCREEN_W || h != UiTheme.SCREEN_H) {
                float sx = w / (float) UiTheme.SCREEN_W;
                float sy = h / (float) UiTheme.SCREEN_H;
                canvas.save();
                canvas.scale(sx, sy);
            }
            try {
                drawCurrentScreen(canvas);
            } catch (Throwable t) {
                // 绘制异常绝不能让应用崩溃（相机上崩溃会导致固件黑屏）
                AppLog.error("界面绘制异常", t);
                canvas.drawColor(Color.BLACK);
                Paint p = UiTheme.newPaint(Color.WHITE, 18f, false);
                canvas.drawText("界面绘制异常：" + t.getClass().getSimpleName(), 20, 40, p);
            }
            if (w != UiTheme.SCREEN_W || h != UiTheme.SCREEN_H) {
                canvas.restore();
            }
        }

        private void drawCurrentScreen(Canvas canvas) {
            switch (mode) {
                case MODE_SETTINGS:
                    settingsScreen.draw(canvas, settings, cameraOk, cameraStatusLine);
                    break;
                case MODE_RUNNING:
                    drawRunScreen(canvas);
                    break;
                case MODE_FINISH_DIALOG:
                    dialogScreen.drawFinishDialog(canvas, finishTakenCount, finishFailedCount, chooseVideo);
                    break;
                case MODE_ERROR_DIALOG:
                    dialogScreen.drawErrorDialog(canvas, errorTitle, errorReason, errorExtra);
                    break;
                case MODE_CONFIRM_DIALOG:
                    dialogScreen.drawConfirmDialog(canvas, confirmTitle, confirmMessage,
                            confirmOption0, confirmOption1, confirmEmphasis);
                    break;
                default:
                    canvas.drawColor(UiTheme.COLOR_BG);
                    break;
            }
        }

        private void drawRunScreen(Canvas canvas) {
            if (scheduler == null) {
                canvas.drawColor(UiTheme.COLOR_BG);
                return;
            }
            // 存储卡剩余空间文本
            String freeText = readFreeText();
            runScreen.draw(canvas, scheduler, scheduler.getSettings(),
                    healthMonitor.getBoxTempText(), freeText,
                    message, messageIsError, previewVisible);
        }
    }

    /** 读取 SD 卡剩余空间并格式化（只为界面显示，读取失败时显示“未知”） */
    private String readFreeText() {
        try {
            android.os.StatFs stat = new android.os.StatFs(
                    android.os.Environment.getExternalStorageDirectory().getAbsolutePath());
            // 与 HealthMonitor 相同：只用 API 10 就存在的 StatFs 方法
            long blockSize = stat.getBlockSize();
            long blocks = stat.getAvailableBlocks();
            long bytes = blockSize * blocks;
            if (bytes <= 0) {
                return "未知";
            }
            long gb = bytes / (1024L * 1024L * 1024L);
            if (gb > 0) {
                return "剩余 " + gb + " GB";
            }
            return "剩余 " + (bytes / (1024L * 1024L)) + " MB";
        } catch (Throwable t) {
            return "未知";
        }
    }

    /* ============================ 与相机固件的协作 ============================ */

    /**
     * 向相机固件申报“当前前台应用”。
     * 广播：com.android.server.DAConnectionManagerService.AppInfoReceive
     * 固件据此判断要不要在自己那一侧启动/停止原生拍摄界面。
     */
    private void notifyAppInfo() {
        try {
            Intent intent = new Intent("com.android.server.DAConnectionManagerService.AppInfoReceive");
            intent.putExtra("package_name", getPackageName());
            intent.putExtra("class_name", getComponentName().getClassName());
            sendBroadcast(intent);
        } catch (Throwable t) {
            AppLog.warn("上报应用信息失败：" + t.getMessage());
        }
    }

    /**
     * 退出应用，返回相机原生拍摄界面。
     * 与索尼官方应用列表中的其他 PMCA 应用一致的做法：
     *   finish() + 启动 ACTION_MAIN/CATEGORY_HOME 的 Intent，
     *   固件会接管并回到原生拍照 UI。
     */
    private void exitToCameraUi() {
        AppLog.info("退出应用，返回相机原生界面");
        stopTaskInternal("已退出", false);
        closeCamera();
        try {
            finish();
            Intent intent = new Intent(Intent.ACTION_MAIN);
            intent.addCategory(Intent.CATEGORY_HOME);
            intent.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            startActivity(intent);
        } catch (Throwable t) {
            AppLog.error("返回相机界面失败", t);
        }
    }

    /* ============================ 电源管理 ============================ */

    /**
     * 获取 CPU 唤醒锁：长间隔（例如 60 秒一张）拍摄时，相机可能因为“省电”把应用挂起，
     * 导致节拍漂移。这里在任务期间持有 PARTIAL_WAKE_LOCK，保证计时与写卡不被中断。
     */
    private void acquireWakeLock() {
        try {
            if (wakeLock == null) {
                PowerManager pm = (PowerManager) getSystemService(POWER_SERVICE);
                if (pm != null) {
                    wakeLock = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "PMCACAP:TIMER");
                    wakeLock.setReferenceCounted(false);
                }
            }
            if (wakeLock != null && !wakeLock.isHeld()) {
                wakeLock.acquire();
            }
        } catch (Throwable t) {
            AppLog.warn("获取唤醒锁失败：" + t.getMessage());
        }
    }

    private void releaseWakeLock() {
        try {
            if (wakeLock != null && wakeLock.isHeld()) {
                wakeLock.release();
            }
        } catch (Throwable ignored) {
        }
    }
}
``````

## 9. `app\src\main\java\com\pmca\intervalcapture\CaptureService.java`

```java
package com.pmca.intervalcapture;

import android.hardware.Camera;
import android.view.SurfaceHolder;

import com.sony.scalar.hardware.CameraEx;

/**
 * CaptureService —— 索尼 PMCA SDK 静态照片捕获封装（本应用最核心的硬件交互层）
 * <p>
 * ===================== 用到的索尼 PMCA SDK 接口清单 =====================
 * 1. {@code CameraEx.open(int cameraId, CameraEx.OpenOptions options)}
 *    打开相机。第一个参数 0 表示主（唯一）成像器；options 传 null 表示使用固件当前设置。
 *    返回值是 CameraEx 实例，本应用所有相机控制都通过它完成。
 *
 * 2. {@code CameraEx.getNormalCamera()} → {@code android.hardware.Camera}
 *    取得“普通静态拍摄”通道。这是拍静态照片必须用的通道。
 *
 * 3. {@code Camera.setPreviewDisplay(SurfaceHolder)} / {@code Camera.startPreview()} /
 *    {@code Camera.stopPreview()}
 *    分别用于绑定预览显示面、开启预览、关闭预览。关闭预览可以显著降低机身发热
 *    （界面上的“预览开关”就是控制这里）。
 *
 * 4. {@code Camera.takePicture(ShutterCallback, PictureCallback, PictureCallback)}
 *    唯一触发静态照片的接口。三个回调全部传 null 的含义是：
 *      - 不使用应用侧 JPEG 回调；
 *      - 照片由相机固件按照“机内图像质量/文件格式（JPG / RAW / JPG+RAW）”设置
 *        直接编码并写入 SD 卡；
 *      - 应用只负责触发，不参与编码，也就不会额外生成视频。
 *    ★ 本应用【绝不】调用 MediaRecorder、也不调用任何视频录制接口。
 *
 * 5. {@code CameraEx.setShutterListener(CameraEx.ShutterListener)}
 *    快门动作回调：onShutter(int status, CameraEx camera)，
 *    status == STATUS_OK 表示快门已正常动作，STATUS_CANCELED 被取消，STATUS_ERROR 出错。
 *
 * 6. {@code CameraEx.setStoreImageCompleteListener(CameraEx.StoreImageCompleteListener)}
 *    写卡完成回调：onDone(int status, CameraEx.StoreImageInfo info, CameraEx camera)，
 *    info.FileName / info.DirectoryName 就是照片在 SD 卡上的文件名与目录名。
 *    这是判断“这一张真的存好了”的唯一可靠依据，间隔拍摄的计数必须以此为准。
 *
 * 7. {@code CameraEx.setErrorCallback(CameraEx.ErrorCallback)}
 *    固件级错误回调（镜头异常、机械错误、写卡失败等）。
 *
 * 8. {@code CameraEx.cancelTakePicture()} —— 用户在写卡过程中按停止键时取消本次拍摄。
 * 9. {@code CameraEx.release()} —— 释放相机（退出应用时必须调用，否则固件侧资源不释放）。
 * 10. {@code CameraEx.setAutoPictureReviewControl(CameraEx.AutoPictureReviewControl)}
 *     传入 null 表示关闭“拍后回看”，避免每张之后屏幕回放照片打断连续拍摄。
 *
 * ===================== 快门/对焦/曝光的锁定策略 =====================
 * 参数“曝光锁定开关”（对应 CameraEx 的 AE/AF/WB 锁定语义）开启时：
 *   1. 首张：正常触发一次自动对焦（固件在 AF 的同时完成 AE 测光与 AWB 白平衡）；
 *   2. 之后每一张：不再调用 autoFocus()，也不再修改任何机内参数，
 *      镜头焦点停在 AF 结果上，AE/AWB 结果被固件沿用，画面亮度保持一致；
 *   3. 因为没有重新测光，连续拍摄的亮度不会出现“忽明忽暗”的闪烁——
 *      这正是间隔拍摄（延时素材）最看重的性质。
 * 关闭时：每张拍摄前都重新 autoFocus()，即每张重新测光对焦（适合光线缓慢变化、
 * 且不希望画面固定在一个曝光值的场景）。
 * 说明：PMCA SDK 未提供“显式 AEL 锁定”的公开方法，本应用通过“不重新测量”的方式
 * 自然保持锁定；真正改变 AE 状态的只有固件自身（例如镜头/光线突变）。
 */
public final class CaptureService {

    /** 相机通道编号：主成像器 */
    private static final int CAMERA_ID = 0;

    private final CaptureCallback callback;

    /** 索尼 CameraEx 实例（打开成功后非空） */
    private CameraEx cameraEx;
    /** 普通静态拍摄通道 */
    private Camera normalCamera;
    /** 是否已经打开了相机 */
    private boolean opened = false;
    /** 预览是否正在运行 */
    private boolean previewRunning = false;
    /** 当前是否有一张正在拍摄中（防止重入） */
    private boolean captureInFlight = false;
    /** 当前这一张是否已经收到“快门动作”回调 */
    private boolean shutterFired = false;

    /** 第一张拍摄后置为 true，用于 AE/AF 锁定策略 */
    private boolean firstShotDone = false;
    /** 本次任务是否需要锁定 AE/AF */
    private boolean lockAeAf = true;

    /** API 10 中 Camera.ShutterCallback 是接口，匿名实现即可（不要用 lambda） */
    private final Camera.ShutterCallback shutterCallback = new Camera.ShutterCallback() {
        @Override
        public void onShutter() {
            shutterFired = true;
            if (callback != null) {
                callback.onShotTaken();
            }
        }
    };

    public CaptureService(CaptureCallback callback) {
        this.callback = callback;
    }

    /* ============================ 打开 / 关闭 ============================ */

    /**
     * 打开相机并（可选）启动预览。
     *
     * @param previewHolder   预览显示面；传 null 表示关闭预览
     * @param lockAeAfOnFirst 是否在第一张之后锁定 AE/AF/WB
     * @return 打开成功返回 true
     */
    public boolean open(SurfaceHolder previewHolder, boolean lockAeAfOnFirst) {
        close();
        this.lockAeAf = lockAeAfOnFirst;
        this.firstShotDone = false;
        try {
            // ① 通过 Sony 无参构造函数拿到 CameraEx 实例，再调用 open 打开相机。
            //    第一个参数 CAMERA_ID=0 表示主（唯一）成像器；
            //    OpenOptions 传 null 表示完全沿用相机固件当前的拍摄设置
            //    （测光模式、AF 模式、画质、JPG/RAW 格式等都由机内菜单决定）。
            //    返回值是一个 CameraEx 实例，之后所有索尼接口都通过它调用。
            cameraEx = new CameraEx().open(CAMERA_ID, null);
            if (cameraEx == null) {
                AppLog.error("CameraEx.open 返回 null");
                return false;
            }
            // ② 取普通静态拍摄通道
            normalCamera = cameraEx.getNormalCamera();
            if (normalCamera == null) {
                AppLog.error("getNormalCamera 返回 null");
                close();
                return false;
            }
            // ③ 注册快门回调：得知每张的快门动作
            cameraEx.setShutterListener(new CameraEx.ShutterListener() {
                @Override
                public void onShutter(int status, CameraEx camera) {
                    handleShutterStatus(status);
                }
            });
            // ④ 注册写卡完成回调：这是“照片真的存进 SD 卡”的唯一依据
            cameraEx.setStoreImageCompleteListener(new CameraEx.StoreImageCompleteListener() {
                @Override
                public void onDone(int status, CameraEx.StoreImageInfo info, CameraEx camera) {
                    handleStoreCompleted(status, info);
                }
            });
            // ⑤ 注册固件级错误回调
            cameraEx.setErrorCallback(new CameraEx.ErrorCallback() {
                @Override
                public void onError(int errorCode, CameraEx camera) {
                    AppLog.error("相机固件错误码：" + errorCode);
                    if (callback != null) {
                        callback.onCameraError(errorCode);
                    }
                }
            });
            // ⑥ 关闭拍后回看：连续拍摄时回放会遮挡画面并拖慢节拍
            try {
                cameraEx.setAutoPictureReviewControl(null);
            } catch (Throwable t) {
                AppLog.warn("关闭拍后回看失败（不影响拍摄）");
            }
            opened = true;
            // ⑦ 按需开启预览
            if (previewHolder != null) {
                startPreview(previewHolder);
            }
            AppLog.info("相机已打开，预览=" + (previewHolder != null));
            if (callback != null) {
                callback.onCameraOpened();
            }
            return true;
        } catch (Throwable t) {
            AppLog.error("打开相机失败", t);
            close();
            return false;
        }
    }

    /**
     * 释放相机。退出运行界面时务必调用，否则相机固件可能一直被占用，
     * 表现为返回原生界面后无法拍照。
     */
    public void close() {
        if (previewRunning && normalCamera != null) {
            try {
                normalCamera.stopPreview();
            } catch (Throwable t) {
                AppLog.warn("stopPreview 异常：" + t.getMessage());
            }
            previewRunning = false;
        }
        if (cameraEx != null) {
            try {
                cameraEx.release();     // 释放固件侧相机资源
            } catch (Throwable t) {
                AppLog.warn("release 异常：" + t.getMessage());
            }
        }
        cameraEx = null;
        normalCamera = null;
        opened = false;
        captureInFlight = false;
        shutterFired = false;
        if (callback != null) {
            callback.onCameraClosed();
        }
    }

    public boolean isOpened() {
        return opened;
    }

    public boolean isCaptureInFlight() {
        return captureInFlight;
    }

    /* ============================ 预览控制 ============================ */

    /** 开启预览：绑定显示面并启动预览流 */
    public boolean startPreview(SurfaceHolder holder) {
        if (normalCamera == null || holder == null) {
            return false;
        }
        try {
            normalCamera.setPreviewDisplay(holder);
            normalCamera.startPreview();
            previewRunning = true;
            return true;
        } catch (Throwable t) {
            AppLog.warn("开启预览失败：" + t.getMessage());
            return false;
        }
    }

    /** 关闭预览（降低发热）；关闭后仍可正常拍照 */
    public void stopPreview() {
        if (normalCamera == null || !previewRunning) {
            return;
        }
        try {
            normalCamera.stopPreview();
        } catch (Throwable t) {
            AppLog.warn("关闭预览异常：" + t.getMessage());
        }
        previewRunning = false;
    }

    public boolean isPreviewRunning() {
        return previewRunning;
    }

    /** 设置显示面（例如从关闭预览切换为打开预览时） */
    public boolean setPreviewDisplay(SurfaceHolder holder) {
        if (normalCamera == null || holder == null) {
            return false;
        }
        try {
            normalCamera.setPreviewDisplay(holder);
            return true;
        } catch (Throwable t) {
            AppLog.warn("绑定预览显示面失败：" + t.getMessage());
            return false;
        }
    }

    /* ============================ 拍照 ============================ */

    /**
     * 触发一张静态照片。
     * <p>
     * 关键实现说明：
     *   - {@code takePicture(shutter, null, null)} 的三个回调全部传 null 时，
     *     PMCA 固件会把照片按机内设置直接写入 SD 卡，应用不参与编码；
     *   - 这里额外传入一个 ShutterCallback，仅用于“知道快门何时动作”，
     *     不会触发任何图片数据回调，也就不会消耗内存；
     *   - 对焦策略：仅在“未锁定 AE/AF”或“第一张”时执行自动对焦；
     *     锁定开启时后续张不再重新对焦，保证焦点与曝光一致。
     *
     * @return true 表示已成功提交拍摄请求
     */
    public boolean takePicture() {
        if (normalCamera == null || !opened) {
            return false;
        }
        if (captureInFlight) {
            // 上一张还没写完卡，拒绝重复触发（间隔太短时会走到这里）
            return false;
        }
        try {
            if (!previewRunning) {
                // 某些固件在“不预览”状态下拍摄前需要一次空预览启动，否则返回错误；
                // 这里用异常捕获兜底，失败也不影响后续 takePicture 的尝试。
                try {
                    normalCamera.startPreview();
                    normalCamera.stopPreview();
                } catch (Throwable ignored) {
                }
            }
            // 对焦策略：未锁定 AE/AF，或者这是第一张时，才重新对焦
            boolean needFocus = (!lockAeAf) || (!firstShotDone);
            if (needFocus) {
                try {
                    // Camera.autoFocus：触发一次单次自动对焦。
                    // 索尼固件在这里会同时完成 AE 与 AWB 的“抓取”，结果在后续张沿用。
                    normalCamera.autoFocus(null);
                } catch (Throwable t) {
                    AppLog.warn("自动对焦触发失败：" + t.getMessage());
                }
                // 给固件一点时间完成测光/对焦（半按快门的等效等待）
                try {
                    Thread.sleep(lockAeAf ? 600 : 250);
                } catch (InterruptedException ignored) {
                }
            }
            captureInFlight = true;
            shutterFired = false;
            // Camera.takePicture：真正触发静态照片写入 SD 卡
            normalCamera.takePicture(shutterCallback, null, null);
            AppLog.info("触发拍摄：第 " + (firstShotDone ? "后续" : "首") + " 张，对焦=" + needFocus);
            return true;
        } catch (Throwable t) {
            captureInFlight = false;
            AppLog.error("takePicture 失败", t);
            return false;
        }
    }

    /**
     * 取消当前正在进行的拍摄（用户按停止键、或写卡超时时调用）。
     */
    public void cancelCapture() {
        if (cameraEx == null || !captureInFlight) {
            return;
        }
        try {
            cameraEx.cancelTakePicture();      // 取消本次拍摄，已写入的部分由固件回滚
            AppLog.info("已取消当前拍摄");
        } catch (Throwable t) {
            AppLog.warn("cancelTakePicture 异常：" + t.getMessage());
        }
        captureInFlight = false;
    }

    /* ============================ 回调处理 ============================ */

    /**
     * 处理快门回调（PMCA SDK 的 ShutterListener，运行在固件线程）。
     */
    private void handleShutterStatus(int status) {
        if (status == CameraEx.ShutterListener.STATUS_OK) {
            // 快门正常动作：照片开始写卡，等待 StoreImageComplete 回调
            AppLog.info("快门动作 OK");
        } else if (status == CameraEx.ShutterListener.STATUS_CANCELED) {
            AppLog.warn("快门被取消");
            captureInFlight = false;
            if (callback != null) {
                callback.onCaptureFailed("本次拍摄被取消");
            }
        } else if (status == CameraEx.ShutterListener.STATUS_ERROR) {
            AppLog.error("快门错误");
            captureInFlight = false;
            if (callback != null) {
                callback.onCaptureFailed("相机快门错误，请检查机身");
            }
        }
    }

    /**
     * 处理写卡完成回调（PMCA SDK 的 StoreImageCompleteListener，运行在固件线程）。
     */
    private void handleStoreCompleted(int status, CameraEx.StoreImageInfo info) {
        captureInFlight = false;
        if (info == null) {
            if (callback != null) {
                callback.onCaptureFailed("拍照失败：未获得文件信息（可能是存储卡已满或故障）");
            }
            return;
        }
        String fileName = info.FileName;
        String dirName = info.DirectoryName;
        if (fileName == null) {
            fileName = "(未知文件名)";
        }
        AppLog.info("照片已保存：" + dirName + "/" + fileName + "，状态=" + status);
        firstShotDone = true;
        if (callback != null) {
            // 注意：status 的具体含义随固件版本不同（0 通常表示成功），
            // 只要拿到了 FileName 就认为写卡完成；失败场景由 ShutterListener / ErrorCallback 报出。
            callback.onImageStored(fileName);
        }
    }

    /* ============================ 状态查询 ============================ */

    /** 本张是否已经收到快门动作回调 */
    public boolean isShutterFired() {
        return shutterFired;
    }

    /** 是否已经拍过第一张（用于界面上提示“已锁定 AE/AF”） */
    public boolean isFirstShotDone() {
        return firstShotDone;
    }
}
``````

## 10. `app\src\main\java\com\pmca\intervalcapture\CaptureSettings.java`

**说明**：参数模型与钳制规则（间隔/张数/格式/曝光锁定/预览）

```java
package com.pmca.intervalcapture;

/**
 * CaptureSettings —— 间隔拍摄参数模型（纯逻辑类，不依赖 Android / 索尼 SDK）
 * <p>
 * 设计要点：
 * 1. 相机硬件限制全部内聚在本类中，任何 UI 层只调用 {@link #adjust} 与格式化方法，
 * 保证“最低 1 秒、步进 0.1 秒、上限 3600 秒、张数 1~9999”等规则只有一处实现；
 * 2. 间隔时间用“十分之一秒”的整数（intervalDeciseconds）保存，避免浮点在相机 JVM
 * 上反复加减产生的 0.1 误差；
 * 3. 本类不接触任何索尼 API，便于在普通电脑上做单元测试。
 * <p>
 * 对应 PMCA SDK 的说明：索尼 PMCA 并没有提供“间隔拍摄”的现成接口，间隔拍摄必须由应用
 * 自己在每一张之间计时，然后反复调用 CameraEx → Camera.takePicture 静态拍摄接口。
 */
public final class CaptureSettings {

    /* ============================ 硬性范围常量 ============================ */

    /** 拍摄间隔下限：0.1 秒为最小步进（A6500 理论最短稳定间隔为 1 秒，低于 1 秒会给出警告） */
    public static final int INTERVAL_MIN_DS = 10;

    /**
     * 拍摄间隔强制下限：1.0 秒 = 10 个 0.1 秒。
     * A6500 机身在连续静态拍摄时，单张的“快门+读出+写卡”至少占用约 0.6~0.9 秒，
     * 因此低于 1.0 秒的间隔在实际使用中不可能达成，此处作为“最低允许设置值”拦截。
     */
    public static final int INTERVAL_HARD_MIN_DS = 10;

    /** 拍摄间隔上限：3600.0 秒（1 小时） */
    public static final int INTERVAL_MAX_DS = 36000;

    /** 固定张数模式下的张数上限 */
    public static final int SHOT_COUNT_MAX = 9999;

    /** 固定张数模式下的张数下限（至少拍 1 张） */
    public static final int SHOT_COUNT_MIN = 1;

    /**
     * 无限模式在内部使用的“虚拟总数”。
     * 用一个大于任何实际可拍数量的数值代替，避免到处写 if (infinite) 分支；
     * 界面显示时把它翻译成“∞”。
     */
    public static final int SHOT_COUNT_INFINITE = 1000000;

    /** 图像格式：仅 JPG。相机固件按机内“影像质量/文件格式”设置执行，本应用不干预。 */
    public static final int FORMAT_JPG_ONLY = 0;
    /** 图像格式：仅 RAW（.ARW）。 */
    public static final int FORMAT_RAW_ONLY = 1;
    /** 图像格式：JPG + RAW 同时保存。 */
    public static final int FORMAT_JPG_PLUS_RAW = 2;

    /** 格式选项的中文名称，供界面直接使用 */
    public static final String[] FORMAT_NAMES = new String[]{"仅 JPG", "仅 RAW(ARW)", "JPG+RAW"};

    /* ============================ 参数字段 ============================ */

    /** 拍摄间隔，单位：0.1 秒 */
    private int intervalDeciseconds = 20;      // 默认 2.0 秒

    /** 是否无限连续拍摄；false 表示“固定张数”模式 */
    private boolean infiniteCount = false;

    /** 固定张数模式下的总张数；无限模式下该值无意义 */
    private int shotCount = 100;               // 默认 100 张

    /** 图像格式，取 FORMAT_xxx 之一 */
    private int imageFormat = FORMAT_JPG_ONLY; // 默认仅 JPG

    /** 是否锁定曝光/对焦/白平衡：开启后第一张锁定，后续复用，防止延时画面明暗闪烁 */
    private boolean lockAeAf = true;           // 默认开启（间隔拍摄的核心需求）

    /** 是否在拍摄过程中显示实时预览；关闭可降低机身发热、提升连续拍摄稳定性 */
    private boolean previewEnabled = true;     // 默认开启

    /* ============================ 读取接口 ============================ */

    public int getIntervalDeciseconds() {
        return intervalDeciseconds;
    }

    /** 返回间隔秒数（浮点），仅用于显示与告警判断，内部计时请使用整数版本 */
    public float getIntervalSeconds() {
        return intervalDeciseconds / 10f;
    }

    /** 间隔毫秒数，供计时器使用 */
    public long getIntervalMillis() {
        return intervalDeciseconds * 100L;
    }

    public boolean isInfiniteCount() {
        return infiniteCount;
    }

    /** 有效总张数：无限模式返回 SHOT_COUNT_INFINITE */
    public int getEffectiveShotCount() {
        return infiniteCount ? SHOT_COUNT_INFINITE : shotCount;
    }

    /** 用户设置的张数（无限模式下仍返回上一次设定的数字，便于切回固定模式） */
    public int getShotCount() {
        return shotCount;
    }

    public int getImageFormat() {
        return imageFormat;
    }

    public boolean isLockAeAf() {
        return lockAeAf;
    }

    public boolean isPreviewEnabled() {
        return previewEnabled;
    }

    /* ============================ 修改接口（带钳制） ============================ */

    /**
     * 上下键调整参数。
     *
     * @param fieldIndex 参数字段序号，见 {@link #FIELD_INTERVAL} 等常量
     * @param direction  正数表示“+”，负数表示“-”
     * @return 实际发生变化时返回 true
     */
    public boolean adjust(int fieldIndex, int direction) {
        switch (fieldIndex) {
            case FIELD_INTERVAL:
                return setIntervalDeciseconds(intervalDeciseconds + direction * intervalStep(direction));
            case FIELD_SHOT_COUNT:
                if (infiniteCount) {
                    // 无限模式下上下键在“无限 / 固定张数”之间切换
                    infiniteCount = false;
                    return true;
                }
                return setShotCount(shotCount + direction * countStep());
            case FIELD_FORMAT:
                return setImageFormat(imageFormat + (direction > 0 ? 1 : -1));
            case FIELD_LOCK:
                lockAeAf = !lockAeAf;
                return true;
            case FIELD_PREVIEW:
                previewEnabled = !previewEnabled;
                return true;
            case FIELD_COUNT_MODE:
                infiniteCount = !infiniteCount;
                return true;
            default:
                return false;
        }
    }

    /** 间隔调整步长：低于 10 秒时每次 0.1 秒，10 秒以上每次 1 秒，方便快速拉到小时级 */
    private int intervalStep(int direction) {
        if (intervalDeciseconds < 100) {
            return 1;           // 0.1 秒
        }
        if (intervalDeciseconds < 600) {
            return 10;          // 1 秒
        }
        return direction > 0 ? 100 : 100; // 10 秒以上每次 10 秒
    }

    /** 张数调整步长：<100 每按一次 ±1，<1000 ±10，其余 ±100 */
    private int countStep() {
        if (shotCount < 100) {
            return 1;
        }
        if (shotCount < 1000) {
            return 10;
        }
        return 100;
    }

    /**
     * 设置间隔（单位 0.1 秒），自动钳制到 [1.0 秒, 3600 秒]。
     * <p>
     * 注意：本类内部表示本身就使用“0.1 秒”为最小单位的整数（10 = 1.0 秒，11 = 1.1 秒），
     * 因此这里只做上下限钳制，不能再做任何“按整秒对齐”的取整，否则 1.1 秒会被错误地变成 1.0 秒。
     */
    public boolean setIntervalDeciseconds(int value) {
        int clamped = clamp(value, INTERVAL_HARD_MIN_DS, INTERVAL_MAX_DS);
        if (clamped == intervalDeciseconds) {
            return false;
        }
        intervalDeciseconds = clamped;
        return true;
    }

    /** 设置固定张数，自动钳制到 [1, 9999] */
    public boolean setShotCount(int value) {
        int clamped = clamp(value, SHOT_COUNT_MIN, SHOT_COUNT_MAX);
        if (clamped == shotCount) {
            return false;
        }
        shotCount = clamped;
        return true;
    }

    /** 设置图像格式（循环切换） */
    public boolean setImageFormat(int value) {
        int count = FORMAT_NAMES.length;
        int v = ((value % count) + count) % count;   // 循环
        if (v == imageFormat) {
            return false;
        }
        imageFormat = v;
        return true;
    }

    public void setLockAeAf(boolean value) {
        lockAeAf = value;
    }

    public void setPreviewEnabled(boolean value) {
        previewEnabled = value;
    }

    public void setInfiniteCount(boolean value) {
        infiniteCount = value;
    }

    /* ============================ 显示格式化 ============================ */

    /** 间隔显示：整数秒显示“2 秒”，带小数显示“2.5 秒” */
    public String formatInterval() {
        int sec = intervalDeciseconds / 10;
        int deci = intervalDeciseconds % 10;
        if (deci == 0) {
            return sec + " 秒";
        }
        return sec + "." + deci + " 秒";
    }

    /** 张数显示：无限模式显示“无限（手动停止）” */
    public String formatShotCount() {
        if (infiniteCount) {
            return "无限（手动停止）";
        }
        return shotCount + " 张";
    }

    public String formatImageFormat() {
        return FORMAT_NAMES[imageFormat];
    }

    public String formatLock() {
        return lockAeAf ? "开（首张后锁定 AE/AF/WB）" : "关（每张重新测光对焦）";
    }

    public String formatPreview() {
        return previewEnabled ? "开" : "关（省电降温）";
    }

    /* ============================ 硬件告警 ============================ */

    /**
     * 判断当前设置是否可能受相机硬件限制而无法严格维持间隔。
     * 触发条件（满足其一即提示）：
     *   - 间隔 < 1.0 秒（A6500 理论上做不到）；
     *   - 间隔 < 1.5 秒且图像格式包含 RAW（RAW 单张写卡时间更长）；
     *   - 间隔 < 1.5 秒且同时开启预览（预览刷新与写卡争抢总线）。
     *
     * @return 需要提示时返回中文提示文本，否则返回 null
     */
    public String getHardwareWarning() {
        float sec = getIntervalSeconds();
        if (sec < 1.0f) {
            return "间隔小于 1 秒，A6500 无法稳定维持";
        }
        if (sec < 1.5f && imageFormat != FORMAT_JPG_ONLY) {
            return "RAW 写卡较慢，1.5 秒以内间隔可能漂移";
        }
        if (sec < 1.5f && previewEnabled) {
            return "预览会占用总线，1.5 秒以内间隔可能漂移";
        }
        return null;
    }

    /* ============================ 参数字段序号 ============================ */

    public static final int FIELD_INTERVAL = 0;
    public static final int FIELD_SHOT_COUNT = 1;
    public static final int FIELD_FORMAT = 2;
    public static final int FIELD_LOCK = 3;
    public static final int FIELD_PREVIEW = 4;
    public static final int FIELD_COUNT_MODE = 5;
    /** 参数总项数，界面按此循环焦点 */
    public static final int FIELD_COUNT = 6;

     * 数值钳制。Android 2.3 的 Math 没有 clamp，这里自己实现，避免依赖高版本 API。
     */
    public static int clamp(int value, int min, int max) {
        if (value < min) {
            return min;
        }
        if (value > max) {
            return max;
        }
        return value;
    }

    /** 深拷贝，防止运行中修改参数影响正在进行的任务 */

        s.shotCount = shotCount;

    public String toPersistLine() {

                + ";infinite=" + infiniteCount
                + ";count=" + shotCount

    public static CaptureSettings fromPersistLine(String line) {
        CaptureSettings s = new CaptureSettings();
        if (line == null) {
            return s;
        }
        String[] parts = line.split(";");
        for (String p : parts) {
            int eq = p.indexOf('=');
            if (eq <= 0) {
                continue;
            }
            String k = p.substring(0, eq).trim();
            String v = p.substring(eq + 1).trim();
            try {
                if (k.equals("interval")) {
                    s.setIntervalDeciseconds(Integer.parseInt(v));

                    s.infiniteCount = "true".equals(v);
                } else if (k.equals("count")) {

                } else if (k.equals("format")) {

                } else if (k.equals("lock")) {
                    s.lockAeAf = "true".equals(v);
                } else if (k.equals("preview")) {
                    s.previewEnabled = "true".equals(v);
                }
            } catch (NumberFormatException ignored) {
                // 单项解析失败时忽略，保留默认值即可
            }
        }
        return s;
    }
``````

## 11. `app\src\main\java\com\pmca\intervalcapture\IntervalScheduler.java`

**说明**：★ 节拍状态机：开始/暂停/继续/停止/溢出检测/倒计时（纯逻辑、可单测）

```java
package com.pmca.intervalcapture;

/**
 * IntervalScheduler —— 间隔拍摄的“节拍状态机”（纯逻辑类，不依赖 Android / 索尼 SDK）
 * <p>
 * 职责：
 *   1. 根据参数与“上一张完成时刻”计算下一次拍摄应该发生在什么时候；
 *   2. 统计已拍张数、失败张数，判断是否达到设定张数（固定张数模式自动停止）；
 *   3. 支持暂停/继续：暂停期间不累计节拍，继续后对齐到“继续时刻 + 间隔”，
 *      且暂停不会重置已拍张数；
 *   4. 处理“拍摄超时”：当一张的曝光时间接近甚至超过设定间隔时，不再等待下一拍而是
 *      立刻连拍（并记录“周期溢出”次数），在界面上提示用户间隔无法保证。
 * <p>
 * 与 PMCA SDK 的关系：索尼没有提供间隔拍摄接口，本类负责“什么时候调用一次
 * CameraEx 的静态拍摄”，真正的调用发生在本类之后由 CaptureService 完成。
 * 相机端计时全部使用 {@link System#nanoTime()}，避免使用 System.currentTimeMillis()
 * 时用户改时间导致节拍跳跃。
 */
public final class IntervalScheduler {

    /** 运行状态 */
    public static final int STATE_IDLE = 0;      // 待机
    public static final int STATE_RUNNING = 1;   // 拍摄中
    public static final int STATE_PAUSED = 2;    // 已暂停
    public static final int STATE_FINISHED = 3;  // 已结束
    public static final int STATE_ERROR = 4;     // 异常停止

    /**
     * 单张拍摄的保护超时（毫秒）。
     * 超过该时间仍未收到相机的“写卡完成”回调，判定为相机异常，主动取消本次拍摄。
     * 之所以取 30 秒：A6500 挂 30 秒长曝光 + RAW 写卡也能在这个时间内完成；
     * 万一机身卡死，应用能及时报错而不是无限等待。
     */
    public static final long CAPTURE_TIMEOUT_MS = 30000L;

    private final CaptureSettings settings;

    private int state = STATE_IDLE;

    /** 已成功写入 SD 卡的张数（以相机 StoreImageComplete 回调为准） */
    private int takenCount = 0;
    /** 拍摄失败张数（快门取消、写卡失败等） */
    private int failedCount = 0;
    /** 因为有张未及时完成，导致无法按设定间隔拍摄的“溢出”次数 */
    private int overrunCount = 0;

    /** 下一张的计划拍摄时刻（nanoTime） */
    private long nextShotAtNanos = 0L;
    /** 当前这一张的开始时刻（nanoTime），用于计算拍摄耗时 */
    private long shotStartNanos = 0L;
    /** 当前这一张是否正在相机中处理（等待回调） */
    private boolean shotInFlight = false;

    /** 最近一张的拍摄+写卡耗时（毫秒），用于界面提示 */
    private long lastShotDurationMs = 0L;

    /** 完成回调（用户界面注册，用于刷新） */
    public interface Listener {
        /** 状态发生变化（开始/暂停/继续/结束/出错） */
        void onStateChanged(int newState);

        /** 成功拍摄并保存一张 */
        void onShotSaved(int takenCount, String fileName);

        /** 拍摄失败 */
        void onShotFailed(String reason);

        /** 达到设定张数，正常结束 */
        void onFinished(int takenCount);
    }

    private Listener listener;

    public IntervalScheduler(CaptureSettings settings) {
        this.settings = settings;
    }

    public void setListener(Listener listener) {
        this.listener = listener;
    }

    public CaptureSettings getSettings() {
        return settings;
    }

    /* ============================ 基本状态查询 ============================ */

    public int getState() {
        return state;
    }

    public int getTakenCount() {
        return takenCount;
    }

    public int getFailedCount() {
        return failedCount;
    }

    public int getOverrunCount() {
        return overrunCount;
    }

    public long getLastShotDurationMs() {
        return lastShotDurationMs;
    }

    public boolean isShotInFlight() {
        return shotInFlight;
    }

    /** 剩余张数（无限模式返回 -1） */
    public int getRemainingCount() {
        if (settings.isInfiniteCount()) {
            return -1;
        }
        int remain = settings.getEffectiveShotCount() - takenCount;
        return remain < 0 ? 0 : remain;
    }

    /** 进度百分比（0~100），无限模式按“已完成张数对 100 取模”显示不了进度，返回 -1 */
    public int getProgressPercent() {
        if (settings.isInfiniteCount()) {
            return -1;
        }
        int total = settings.getEffectiveShotCount();
        if (total <= 0) {
            return 0;
        }
        return (int) (takenCount * 100L / total);
    }

    /* ============================ 生命周期操作 ============================ */

    /**
     * 开始拍摄（待机 → 拍摄中）。第一张立即拍摄，之后按间隔计时。
     *
     * @return 是否真的切换了状态
     */
    public boolean start() {
        if (state != STATE_IDLE && state != STATE_FINISHED && state != STATE_ERROR) {
            return false;
        }
        takenCount = 0;
        failedCount = 0;
        overrunCount = 0;
        lastShotDurationMs = 0L;
        shotInFlight = false;
        nextShotAtNanos = System.nanoTime();   // 立即拍第一张
        setState(STATE_RUNNING);
        return true;
    }

    /**
     * 暂停（拍摄中 → 已暂停）。
     * 暂停不会重置计数；如果暂停时有一张还在相机里处理，等它回调后再真正进入暂停态，
     * 这里只负责切换状态，剩余的收尾交给 onShotDone 处理。
     */
    public boolean pause() {
        if (state != STATE_RUNNING) {
            return false;
        }
        setState(STATE_PAUSED);
        return true;
    }

    /**
     * 继续（已暂停 → 拍摄中）。继续后从当前时刻重新开始计时，不补拍暂停期间缺的张数。
     */
    public boolean resume() {
        if (state != STATE_PAUSED) {
            return false;
        }
        // 注意单位：nanoTime() 是纳秒，getIntervalMillis() 是毫秒，必须乘 1e6 换算，
        // 否则“继续”之后会把下一张的触发时刻设成一个极小的纳秒值，导致立即连拍。
        nextShotAtNanos = System.nanoTime() + settings.getIntervalMillis() * 1000000L;
        setState(STATE_RUNNING);
        return true;
    }

    /**
     * 停止拍摄并结束任务（用户按“下”键或发生错误时调用）。
     *
     * @param reason 结束原因，用于界面显示
     */
    public void stop(String reason) {
        if (state == STATE_FINISHED || state == STATE_ERROR || state == STATE_IDLE) {
            return;
        }
        setState(STATE_FINISHED);
        if (listener != null) {
            listener.onFinished(takenCount);
        }
    }

    /** 发生不可恢复错误（SD 卡满 / 写卡失败 / 过热）时由监控模块调用 */
    public void fail(String reason) {
        if (state == STATE_ERROR) {
            return;
        }
        setState(STATE_ERROR);
        if (listener != null) {
            listener.onShotFailed(reason);
        }
    }

    /* ============================ 节拍推进 ============================ */

    /**
     * 判断当前是否应该触发一次拍摄。
     * 由界面主循环（约每 100ms 调用一次）调用。
     *
     * @return true 表示可以调用相机拍照接口
     */
    public boolean shouldTriggerNow() {
        if (state != STATE_RUNNING || shotInFlight) {
            return false;
        }
        long now = System.nanoTime();
        return now >= nextShotAtNanos;
    }

    /**
     * 标记“已经调用相机拍照接口”，开始计时本张的拍摄时长。
     */
    public void onShotStarted() {
        shotStartNanos = System.nanoTime();
        shotInFlight = true;
    }

    /**
     * 相机报告本张已经成功写入 SD 卡（StoreImageComplete 回调）。
     *
     * @param fileName 相机给出的文件名，例如 "DSC01234.JPG"
     */
    public void onShotSaved(String fileName) {
        if (!shotInFlight) {
            return;
        }
        shotInFlight = false;
        takenCount++;
        lastShotDurationMs = (System.nanoTime() - shotStartNanos) / 1000000L;
        scheduleNext();
        if (listener != null) {
            listener.onShotSaved(takenCount, fileName);
        }
        if (!settings.isInfiniteCount() && takenCount >= settings.getEffectiveShotCount()) {
            setState(STATE_FINISHED);
            if (listener != null) {
                listener.onFinished(takenCount);
            }
        }
    }

    /**
     * 相机报告本张失败（快门被取消 / 写卡失败 / 曝光异常）。
     * 失败张数不计入已拍数量，但同样进入下一拍节拍，避免一直卡在同一张。
     */
    public void onShotFailed(String reason) {
        if (!shotInFlight) {
            return;
        }
        shotInFlight = false;
        failedCount++;
        lastShotDurationMs = (System.nanoTime() - shotStartNanos) / 1000000L;
        scheduleNext();
        if (listener != null) {
            listener.onShotFailed(reason);
        }
    }

    /**
     * 计算下一张的计划时刻：
     *   - 正常情况下：本张完成时刻 + 间隔（这样每张之间的“完成→开始”间隔严格等于设定值）；
     *   - 若本张耗时已经超过设定间隔（例如长曝光 3 秒但间隔设 1 秒），
     *     则立刻开始下一张，并累加溢出计数，让界面能提示“间隔无法严格维持”。
     * 无限模式与固定张数模式使用完全相同的规则。
     */
    private void scheduleNext() {
        long intervalMs = settings.getIntervalMillis();
        if (lastShotDurationMs >= intervalMs) {
            overrunCount++;
            // 本张耗时已超过间隔：上一张刚开始后 intervalMs 的那一拍已经被“吃掉”，
            // 直接按“本张开始 + 间隔”排下去即可；若这个时刻也已过去则立刻连拍。
            nextShotAtNanos = shotStartNanos + intervalMs * 1000000L;
            if (nextShotAtNanos < System.nanoTime()) {
                nextShotAtNanos = System.nanoTime();
            }
        } else {
            // 正常情况：下一张在“本张完成时刻 + 间隔”触发
            nextShotAtNanos = shotStartNanos + (lastShotDurationMs + intervalMs) * 1000000L;
        }
    }

    /**
     * 距下一次拍摄的倒计时（毫秒），用于界面显示“距下一张 x.x 秒”。
     * 暂停或未运行时返回 -1。
     */
    public long getCountdownMillis() {
        if (state != STATE_RUNNING) {
            return -1;
        }
        long now = System.nanoTime();
        long diff = nextShotAtNanos - now;
        if (diff < 0) {
            return 0;
        }
        return diff / 1000000L;
    }

    /**
     * 检查单张拍摄是否超时（相机回调迟迟不来）。
     *
     * @return true 表示已超时，调用方应提示错误并停止任务
     */
    public boolean isCaptureTimeout() {
        return shotInFlight && (System.nanoTime() - shotStartNanos) / 1000000L > CAPTURE_TIMEOUT_MS;
    }

    /**
     * 预计剩余时长（秒），用于开始前提示用户。无限模式返回 -1。
     */
    public long getEstimatedTotalSeconds() {
        if (settings.isInfiniteCount()) {
            return -1;
        }
        long perShot = Math.max(settings.getIntervalMillis(), 1000L);
        return settings.getEffectiveShotCount() * perShot / 1000L;
    }

    private void setState(int newState) {
        if (state == newState) {
            return;
        }
        state = newState;
        if (listener != null) {
            listener.onStateChanged(newState);
        }
    }

    /** 状态中文名称，供界面直接显示 */
    public static String stateName(int state) {
        switch (state) {
            case STATE_IDLE:
                return "待机";
            case STATE_RUNNING:
                return "拍摄中";
            case STATE_PAUSED:
                return "已暂停";
            case STATE_FINISHED:
                return "已结束";
            case STATE_ERROR:
                return "异常停止";
            default:
                return "未知";
        }
    }
}
``````

```java
package com.pmca.intervalcapture;

/**
 * HealthMonitor —— 拍摄过程中的健康巡检（SD 卡 / 温度 / 电池 / 存储写入）
 * <p>
 * 需求映射：“SD 卡满、存储卡读写失败、相机过热，立刻停止拍摄并弹出中文报错”。
 * <p>
 * 用到的接口：
 *   1. SD 卡剩余空间：android.os.StatFs（平台基础 API，读取挂载点文件系统信息）——
 *      这是相机 SD 卡在系统里的真实挂载点，读取不需要任何索尼私有接口，
 *      也不涉及“用电脑/手机写卡”，照片写入仍完全由相机固件完成；
 *   2. 相机温度：com.sony.scalar.sysutil.didep.Temperature（索尼私有工具类）
 *      - new Temperature() 创建实例；
 *      - getStatus() 返回 Temperature.Status；
 *      - Status.get(Status.ID_BOX) 取机身温度等级，Status.get(Status.ID_BATTERY) 取电池温度等级；
 *      - 等级含义：STAT_NORMAL(1) 正常、STAT_PREHIGH(2) 偏高、STAT_NEARHIGH(3) 接近过热限、
 *        STAT_HIGH(4) 过热（此时固件可能拒绝拍摄）；
 *      - release() 释放实例（必须调用，否则会泄漏固件侧资源）。
 *   3. 温度状态回调：Temperature#setStatusCallback 可在温度等级变化时主动通知，
 *      这里用轮询 + 回调双保险，回调用不到时轮询仍能发现。
 *   4. 电池：com.sony.scalar.sysutil.didep.Power（可选，仅用于提示，不作为停止条件）。
 * <p>
 * 安全检查策略（宁可提前停下，也不要拍到一半没电/满卡）：
 *   - 剩余空间 < 100 MB：警告；
 *   - 剩余空间 < 30 MB 或写卡连续失败：立即停止；
 *   - 机身温度等级 >= STAT_NEARHIGH：警告；>= STAT_HIGH：立即停止；
 *   - 单张写卡失败累计 3 次：立即停止（判断为卡故障）。
 */
public final class HealthMonitor {

    /** 立即停止的原因（中文，直接显示到屏幕） */
    private String fatalReason = null;
    /** 警告信息（中文），界面用黄色显示 */
    private String warningText = null;

    /** 上次检查的序号，用于控制检查频率（每 10 次拍照检查一次，避免频繁访问固件） */
    private int checkCounter = 0;
    /** 连续写卡失败次数 */
    private int consecutiveStoreFailures = 0;

    /** 阈值：剩余空间警告 100MB */
    private static final long WARN_FREE_BYTES = 100L * 1024 * 1024;
    /** 阈值：剩余空间低于 30MB 必须停止（一张 A6500 RAW 约 24MB） */
    private static final long STOP_FREE_BYTES = 30L * 1024 * 1024;
    /** 连续写卡失败上限 */
    private static final int MAX_CONSECUTIVE_FAILURES = 3;

    /** 索尼温度实例；创建失败（例如其他机型固件无此类）时为 null，功能自动降级 */
    private com.sony.scalar.sysutil.didep.Temperature temperature;
    private boolean temperatureAvailable = false;

    /** 最近的温度等级（1~4），用于界面显示 */
    private int boxTempLevel = com.sony.scalar.sysutil.didep.Temperature.Status.STAT_UNKNOWN;

    /**
     * 温度状态回调占位（本应用不注册回调）。
     * <p>
     * 说明：Temperature.setStatusCallback 需要的是固件内部定义的类型
     * Temperature$StatusCallback，其抽象方法名各固件版本不同、公开资料无法确定，
     * 因此这里不注册回调，改为在拍摄循环中按固定频率主动轮询 getStatus()，
     * 效果等价，又不会因为方法名不匹配在运行时抛 AbstractMethodError 打断拍摄。
     */
    private static final com.sony.scalar.sysutil.didep.Temperature.StatusCallback TEMP_CALLBACK = null;

    /** 打开温度监控（进入运行界面时调用） */
    public void open() {
        fatalReason = null;
        warningText = null;
        consecutiveStoreFailures = 0;
        boxTempLevel = com.sony.scalar.sysutil.didep.Temperature.Status.STAT_UNKNOWN;
        try {
            // 创建索尼温度管理实例（com.sony.scalar.sysutil.didep.Temperature）。
            // 该对象背后连接的是相机固件的温度传感器服务，创建后需要 release() 释放。
            temperature = new com.sony.scalar.sysutil.didep.Temperature();
            // 主动轮询模式：不注册回调（见 tempCallback 处的说明），
            // 在每次 checkBeforeShot 时调用 getStatus() 读取最新等级。
            temperatureAvailable = true;
            AppLog.info("温度监控已启动（轮询模式）");
        } catch (Throwable t) {
            // 某些固件版本没有 Temperature 类，或权限不足：降级为“无温度监控”，不阻塞拍摄
            temperature = null;
            temperatureAvailable = false;
            AppLog.warn("温度监控不可用：" + t.getClass().getSimpleName());
        }
    }

    /** 关闭温度监控（退出运行界面时必须调用，避免固件侧资源泄漏） */
    public void close() {
        if (temperature != null) {
            try {
                temperature.release();   // 释放固件侧温度服务句柄
            } catch (Throwable t) {
                AppLog.warn("释放温度监控失败：" + t.getMessage());
            }
            temperature = null;
        }
        temperatureAvailable = false;
    }

    public boolean isTemperatureAvailable() {
        return temperatureAvailable;
    }

    public int getBoxTempLevel() {
        return boxTempLevel;
    }

    /** 温度等级的中文描述 */
    public String getBoxTempText() {
        if (!temperatureAvailable) {
            return "不可用";
        }
        switch (boxTempLevel) {
            case com.sony.scalar.sysutil.didep.Temperature.Status.STAT_NORMAL:
                return "正常";
            case com.sony.scalar.sysutil.didep.Temperature.Status.STAT_PREHIGH:
                return "偏高";
            case com.sony.scalar.sysutil.didep.Temperature.Status.STAT_NEARHIGH:
                return "接近过热";
            case com.sony.scalar.sysutil.didep.Temperature.Status.STAT_HIGH:
                return "过热";
            default:
                return "未知";
        }
    }

    public String getFatalReason() {
        return fatalReason;
    }

    public String getWarningText() {
        return warningText;
    }

    public boolean isFatal() {
        return fatalReason != null;
    }

    /* ============================ 巡检入口 ============================ */

    /**
     * 每次“准备拍下一张”之前调用。
     *
     * @param isFirstShot 是否是本次任务的第一张（第一张前必须做完整检查）
     * @return true 表示安全，可以拍；false 表示已经设置 fatalReason，应立刻停止
     */
    public boolean checkBeforeShot(boolean isFirstShot) {
        fatalReason = null;
        warningText = null;

        // 1) 每次都检查：SD 卡是否存在且可写
        String storageState = readStorageState();
        if (!"mounted".equals(storageState)) {
            fatalReason = storageStateToChinese(storageState);
            return false;
        }

        // 2) 每次都检查：剩余空间
        long free = readFreeBytes();
        if (free >= 0) {
            if (free < STOP_FREE_BYTES) {
                fatalReason = "存储卡空间不足（剩余 " + (free / 1024 / 1024) + "MB），已停止拍摄";
                return false;
            } else if (free < WARN_FREE_BYTES) {
                warningText = "存储卡剩余空间仅 " + (free / 1024 / 1024) + "MB";
            }
        }

        // 3) 降低频率检查温度：每 5 张或第一张之前检查一次，减少对固件的调用压力
        if (isFirstShot || (++checkCounter >= 5)) {
            checkCounter = 0;
            if (!checkTemperature()) {
                return false;
            }
        }
        return true;
    }

    /**
     * 检查温度。返回 false 表示过热必须停止。
     */
    private boolean checkTemperature() {
        if (temperature == null) {
            return true;
        }
        try {
            com.sony.scalar.sysutil.didep.Temperature.Status status = temperature.getStatus();
            if (status == null) {
                return true;
            }
            // ID_BOX = 机身整体温度，是连续拍摄最主要的热源指标
            boxTempLevel = status.get(com.sony.scalar.sysutil.didep.Temperature.Status.ID_BOX);
            if (boxTempLevel >= com.sony.scalar.sysutil.didep.Temperature.Status.STAT_HIGH) {
                fatalReason = "相机过热，已自动停止拍摄。请关闭电源待机身降温后再拍摄";
                return false;
            }
            if (boxTempLevel >= com.sony.scalar.sysutil.didep.Temperature.Status.STAT_NEARHIGH) {
                // 接近过热：不立即停止，但建议用户关闭预览、延长间隔
                warningText = "机身温度接近上限，建议关闭预览或加大间隔";
            }
        } catch (Throwable t) {
            // 温度接口异常时不影响拍摄，只记录日志
            AppLog.warn("读取温度失败：" + t.getClass().getSimpleName());
        }
        return true;
    }

    /**
     * 相机报告“写卡完成”时调用，清零连续失败计数。
     */
    public void onStoreSuccess() {
        consecutiveStoreFailures = 0;
    }

    /**
     * 相机报告“写卡失败/快门异常”时调用。
     *
     * @return true 表示连续失败次数过多，应停止拍摄
     */
    public boolean onStoreFailure() {
        consecutiveStoreFailures++;
        if (consecutiveStoreFailures >= MAX_CONSECUTIVE_FAILURES) {
            fatalReason = "存储卡读写失败（连续 " + consecutiveStoreFailures
                    + " 次），请检查存储卡或更换存储卡后重试";
            return true;
        }
        return false;
    }

    /* ============================ 平台信息读取 ============================ */

    /** 读取 SD 卡挂载状态（平台 API） */
    private String readStorageState() {
        try {
            return android.os.Environment.getExternalStorageState();
        } catch (Throwable t) {
            return "unknown";
        }
    }

    private String storageStateToChinese(String state) {
        if ("removed".equals(state)) {
            return "未检测到存储卡，请插入存储卡后重试";
        }
        if ("mounted_ro".equals(state)) {
            return "存储卡处于只读状态（写保护），无法保存照片";
        }
        if ("unmounted".equals(state) || "bad_removal".equals(state)) {
            return "存储卡已拔出或未正确挂载，已停止拍摄";
        }
        if ("unmountable".equals(state) || "checking".equals(state)) {
            return "存储卡正在检查或无法挂载，已停止拍摄";
        }
        return "存储卡不可用（" + state + "），已停止拍摄";
    }

    /** 读取 SD 卡剩余字节数；失败返回 -1（此时不做空间判断） */
    private long readFreeBytes() {
        try {
            android.os.StatFs stat = new android.os.StatFs(
                    android.os.Environment.getExternalStorageDirectory().getAbsolutePath());
            // getBlockSize() / getAvailableBlocks() 在 API 10（Android 2.3）就存在，
            // 全系列相机固件通用，因此不使用后来才出现的 getAvailableBytes()。
            long blockSize = stat.getBlockSize();
            long availBlocks = stat.getAvailableBlocks();
            if (blockSize <= 0 || availBlocks < 0) {
                return -1;
            }
            return blockSize * availBlocks;
        } catch (Throwable t) {
            return -1;
        }
    }
}
``````

```java
package com.pmca.intervalcapture;

import android.view.KeyEvent;

import com.sony.scalar.sysutil.ScalarInput;

/**
 * CameraKeyRouter —— 相机实体按键路由
 * <p>
 * A6500 机身虽有触摸屏，但本应用**不做任何触摸交互**，全部操作都靠机身实体按键；
 * 这样在不同 PMCA 机型（A6000 系无触屏 / A6500 有触屏）上行为完全一致，也避免误触。
 * 相机运行时把按键事件以 {@link KeyEvent} 形式
 * 投递到 Activity，但普通的 keyCode 在相机上不可靠（同一物理键在不同机型上 keyCode 可能
 * 不同），真正稳定的是“扫描码” {@link KeyEvent#getScanCode()}。
 * 扫描码的取值定义在索尼私有类 {@code com.sony.scalar.sysutil.ScalarInput} 中，
 * 例如：
 * <pre>
 *   ScalarInput.ISV_KEY_UP      = 103   上方向键
 *   ScalarInput.ISV_KEY_DOWN    = 108   下方向键
 *   ScalarInput.ISV_KEY_LEFT    = 105   左方向键
 *   ScalarInput.ISV_KEY_RIGHT   = 106   右方向键
 *   ScalarInput.ISV_KEY_ENTER   = 232   中央确认键
 *   ScalarInput.ISV_KEY_MENU    = 514   MENU 键（在应用内通常等价于“返回相机”）
 *   ScalarInput.ISV_KEY_SK1     = 229   左上软键
 *   ScalarInput.ISV_KEY_SK2     = 513   右下软键（部分机型为“删除/垃圾桶”键）
 *   ScalarInput.ISV_KEY_FN      = 520   FN 键
 *   ScalarInput.ISV_KEY_AEL     = 532   AEL 曝光锁定键
 *   ScalarInput.ISV_KEY_S1_1    = 516   快门半按
 *   ScalarInput.ISV_KEY_S2      = 518   快门全按
 * </pre>
 * 本类把这些扫描码翻译成“应用动作”，这样界面层只处理语义动作，与具体按键彻底解耦。
 */
public final class CameraKeyRouter {

    /* ============================ 应用动作定义 ============================ */

    /** 无动作 */
    public static final int ACTION_NONE = 0;
    /** 上：参数页 = 上一个参数；运行页 = 无（或加长间隔） */
    public static final int ACTION_UP = 1;
    /** 下：参数页 = 下一个参数；运行页 = 停止拍摄并返回参数页 */
    public static final int ACTION_DOWN = 2;
    /** 左：参数减 */
    public static final int ACTION_LEFT = 3;
    /** 右：参数加 */
    public static final int ACTION_RIGHT = 4;
    /** 确认：参数页 = 开始拍摄 / 运行页 = 开始-暂停切换 / 弹窗 = 确认选项 */
    public static final int ACTION_ENTER = 5;
    /** 菜单/返回：参数页 = 退出应用；运行页 = 停止并退出；弹窗 = 选“否” */
    public static final int ACTION_BACK = 6;
    /** FN 键：参数页 = 循环切换焦点（与上下键同效，作为备用） */
    public static final int ACTION_FN = 7;
    /** 快门键：参数页 = 直接开始；运行页 = 暂停/继续 */
    public static final int ACTION_SHUTTER = 8;
    /** 播放键：参数页 = 显示本次统计信息 */
    public static final int ACTION_PLAY = 9;

    /** 动作文本（界面底部按键提示直接使用） */
    public static String actionName(int action) {
        switch (action) {
            case ACTION_UP:
                return "上";
            case ACTION_DOWN:
                return "下";
            case ACTION_LEFT:
                return "左";
            case ACTION_RIGHT:
                return "右";
            case ACTION_ENTER:
                return "确认";
            case ACTION_BACK:
                return "菜单";
            case ACTION_FN:
                return "FN";
            case ACTION_SHUTTER:
                return "快门";
            case ACTION_PLAY:
                return "播放";
            default:
                return "?";
        }
    }

    private CameraKeyRouter() {
    }

    /* ============================ 按键 → 动作 ============================ */

    /**
     * 把一次“按下”的 KeyEvent 翻译成应用动作。
     *
     * @param event 按键事件
     * @return 应用动作常量
     */
    public static int routeKeyDown(KeyEvent event) {
        if (event == null) {
            return ACTION_NONE;
        }
        int scanCode = event.getScanCode();
        // 部分老固件扫描码为 0，此时退化为标准 keyCode 判断
        if (scanCode == 0) {
            return routeKeyCode(event.getKeyCode());
        }
        switch (scanCode) {
            case ScalarInput.ISV_KEY_UP:
            case ScalarInput.ISV_KEY_IR_UP:
                return ACTION_UP;
            case ScalarInput.ISV_KEY_DOWN:
            case ScalarInput.ISV_KEY_IR_DOWN:
                return ACTION_DOWN;
            case ScalarInput.ISV_KEY_LEFT:
            case ScalarInput.ISV_KEY_IR_LEFT:
                return ACTION_LEFT;
            case ScalarInput.ISV_KEY_RIGHT:
            case ScalarInput.ISV_KEY_IR_RIGHT:
                return ACTION_RIGHT;
            case ScalarInput.ISV_KEY_ENTER:
            case ScalarInput.ISV_KEY_SK1:
                return ACTION_ENTER;
            case ScalarInput.ISV_KEY_MENU:
            case ScalarInput.ISV_KEY_SK2:
            case ScalarInput.ISV_KEY_DELETE:
                return ACTION_BACK;
            case ScalarInput.ISV_KEY_FN:
                return ACTION_FN;
            case ScalarInput.ISV_KEY_S2:
                return ACTION_SHUTTER;
            case ScalarInput.ISV_KEY_S1_1:
            case ScalarInput.ISV_KEY_S1_2:
                // 半按快门不做动作：本应用自己控制对焦
                return ACTION_NONE;
            case ScalarInput.ISV_KEY_PLAY:
                return ACTION_PLAY;
            default:
                return ACTION_NONE;
        }
    }

    /**
     * 把“抬起”的 KeyEvent 翻译成应用动作。
     * 应用采用“按下即触发”的交互方式（相机 UI 习惯），抬起事件一律吞掉，
     * 避免同一个键产生两次动作。
     */
    public static int routeKeyUp(KeyEvent event) {
        return ACTION_NONE;
    }

    /** 扫描码缺失时的兜底：使用标准 Android keyCode */
    private static int routeKeyCode(int keyCode) {
        switch (keyCode) {
            case KeyEvent.KEYCODE_DPAD_UP:
                return ACTION_UP;
            case KeyEvent.KEYCODE_DPAD_DOWN:
                return ACTION_DOWN;
            case KeyEvent.KEYCODE_DPAD_LEFT:
                return ACTION_LEFT;
            case KeyEvent.KEYCODE_DPAD_RIGHT:
                return ACTION_RIGHT;
            case KeyEvent.KEYCODE_DPAD_CENTER:
            case KeyEvent.KEYCODE_ENTER:
                return ACTION_ENTER;
            case KeyEvent.KEYCODE_BACK:
            case KeyEvent.KEYCODE_MENU:
                return ACTION_BACK;
            case KeyEvent.KEYCODE_CAMERA:
                return ACTION_SHUTTER;
            default:
                return ACTION_NONE;
        }
    }

    /**
     * 该扫描码是否属于“本应用要消费”的按键。
     * 返回 true 时 Activity 会返回 true 阻止事件继续传给系统（否则固件可能同时执行
     * “返回原生界面”之类的默认行为，导致两个动作叠加）。
     */
    public static boolean isConsumed(KeyEvent event) {
        return routeKeyDown(event) != ACTION_NONE;
    }
}
``````

```java
package com.pmca.intervalcapture;

/**
 * SettingsStore —— 参数与统计数据的持久化
 * <p>
 * 用途：
 *   1. 记住用户上次设置的间隔、张数、格式等，下次进入应用不用重设；
 *   2. 断电/异常退出后能恢复“上次拍到哪里”（仅用于显示，不影响 SD 卡上的原图）。
 * <p>
 * 实现选择：使用 Android 的 SharedPreferences（平台基础能力，不是相机接口）。
 * 之所以同时把参数写一份到 SD 卡（PMCACAP/SETTINGS.TXT）：
 *   相机固件有时会因为刷入/升级/恢复出厂而清空应用私有数据，
 *   而 SD 卡上的文本文件可以手工查看，方便排查“参数怎么变回默认值了”。
 */
public final class SettingsStore {

    private static final String PREF_NAME = "interval_capture_pref";
    private static final String KEY_SETTINGS = "settings_line";
    private static final String KEY_TOTAL_TAKEN = "total_taken";

    private final android.content.Context context;

    public SettingsStore(android.content.Context context) {
        this.context = context;
    }

    /** 读取上次保存的参数；没有记录时返回默认参数 */
    public CaptureSettings load() {
        try {
            android.content.SharedPreferences sp =
                    context.getSharedPreferences(PREF_NAME, android.content.Context.MODE_PRIVATE);
            String line = sp.getString(KEY_SETTINGS, null);
            if (line == null) {
                // 私有数据被清空时，尝试从 SD 卡恢复
                line = readSdSettings();
            }
            return CaptureSettings.fromPersistLine(line);
        } catch (Throwable t) {
            AppLog.error("读取参数失败，使用默认值", t);
            return new CaptureSettings();
        }
    }

    /** 保存参数 */
    public void save(CaptureSettings settings) {
        try {
            android.content.SharedPreferences sp =
                    context.getSharedPreferences(PREF_NAME, android.content.Context.MODE_PRIVATE);
            String line = settings.toPersistLine();
            sp.edit().putString(KEY_SETTINGS, line).commit();
            writeSdSettings(line);
        } catch (Throwable t) {
            AppLog.error("保存参数失败", t);
        }
    }

    /** 累计拍摄张数（跨多次任务累加，仅供用户参考） */
    public int getTotalTaken() {
        try {
            return context.getSharedPreferences(PREF_NAME, android.content.Context.MODE_PRIVATE)
                    .getInt(KEY_TOTAL_TAKEN, 0);
        } catch (Throwable t) {
            return 0;
        }
    }

    /** 累加拍摄张数 */
    public void addTotalTaken(int count) {
        try {
            android.content.SharedPreferences sp =
                    context.getSharedPreferences(PREF_NAME, android.content.Context.MODE_PRIVATE);
            int total = sp.getInt(KEY_TOTAL_TAKEN, 0) + count;
            sp.edit().putInt(KEY_TOTAL_TAKEN, total).commit();
        } catch (Throwable t) {
            AppLog.error("累计张数写入失败", t);
        }
    }

    /* ------------------------- SD 卡冗余备份 ------------------------- */

    private java.io.File sdSettingsFile() {
        try {
            java.io.File dir = new java.io.File(
                    android.os.Environment.getExternalStorageDirectory().getAbsolutePath(),
                    AppLog.SD_DIR_NAME);
            if (!dir.exists()) {
                dir.mkdirs();
            }
            return new java.io.File(dir, "SETTINGS.TXT");
        } catch (Throwable t) {
            return null;
        }
    }

    private String readSdSettings() {
        java.io.File f = sdSettingsFile();
        if (f == null || !f.exists()) {
            return null;
        }
        java.io.BufferedReader br = null;
        try {
            br = new java.io.BufferedReader(new java.io.InputStreamReader(
                    new java.io.FileInputStream(f), "UTF-8"));
            return br.readLine();
        } catch (Throwable t) {
            return null;
        } finally {
            if (br != null) {
                try {
                    br.close();
                } catch (Throwable ignored) {
                }
            }
        }
    }

    private void writeSdSettings(String line) {
        java.io.File f = sdSettingsFile();
        if (f == null) {
            return;
        }
        java.io.FileOutputStream fos = null;
        try {
            fos = new java.io.FileOutputStream(f, false);
            fos.write((line + "\r\n").getBytes("UTF-8"));
        } catch (Throwable t) {
            AppLog.warn("SD 卡参数备份写入失败：" + t.getMessage());
        } finally {
            if (fos != null) {
                try {
                    fos.close();
                } catch (Throwable ignored) {
                }
            }
        }
    }
}
``````

```java
package com.pmca.intervalcapture;

/**
 * AppLog —— 轻量日志工具
 * <p>
 * 为什么要写日志到 SD 卡：
 *   1. 相机屏幕只有 640x480，出错时能显示的信息有限；
 *   2. 相机没有 USB 调试口可直接看 logcat（需要 PMCA-RE 的 tweak 开启 adb/telnet）,
 *      把关键事件写到 /mnt/sdcard/PMCACAP/LOG.TXT 后，把卡插到电脑上就能看到；
 *   3. 长时间无人值守拍摄结束后，用户可以据此核对每一张的触发时间与耗时。
 * <p>
 * 涉及的非索尼 API：java.io / android.util.Log，均为平台基础类，不涉及任何安卓专属
 * 相机控制函数（相机控制一律走 com.sony.scalar.*）。
 */
public final class AppLog {

    private static final String TAG = "PMCACAP";

    /** SD 卡上的日志目录（相机 SD 卡的挂载点由 Environment 提供） */
    public static final String SD_DIR_NAME = "PMCACAP";

    private static java.io.File logFile;
    private static boolean sdWritable = true;
    /** 日志文件超过该大小时截断重写，避免长时间拍摄把卡写满 */
    private static final long MAX_LOG_BYTES = 512 * 1024L;

    private AppLog() {
    }

    /** 初始化：尝试创建 SD 卡日志目录；失败不影响主流程（只打 logcat） */
    public static void init(String sdRootPath) {
        try {
            java.io.File dir = new java.io.File(sdRootPath, SD_DIR_NAME);
            if (!dir.exists()) {
                sdWritable = dir.mkdirs();
            }
            logFile = new java.io.File(dir, "LOG.TXT");
        } catch (Throwable t) {
            sdWritable = false;
            logFile = null;
        }
    }

    public static String getLogFilePath() {
        return logFile == null ? "(未启用)" : logFile.getAbsolutePath();
    }

    public static void info(String msg) {
        log("I", msg);
    }

    public static void warn(String msg) {
        log("W", msg);
    }

    public static void error(String msg) {
        log("E", msg);
    }

    /** 记录异常堆栈（只记录类名+消息，避免长栈写满卡） */
    public static void error(String msg, Throwable t) {
        if (t == null) {
            log("E", msg);
            return;
        }
        log("E", msg + " | " + t.getClass().getName() + ": " + t.getMessage());
    }

    private static void log(String level, String msg) {
        String line = "[" + level + "] " + System.currentTimeMillis() + " " + msg;
        // 1) 输出到 logcat，方便通过 PMCA-RE 开启 adb 后排查
        try {
            android.util.Log.println(level.equals("E") ? android.util.Log.ERROR
                    : level.equals("W") ? android.util.Log.WARN : android.util.Log.INFO, TAG, msg);
        } catch (Throwable ignored) {
        }
        // 2) 追加写入 SD 卡
        if (!sdWritable || logFile == null) {
            return;
        }
        if (logFile.length() > MAX_LOG_BYTES) {
            // 超限：清空重来，保证不会把卡写满导致拍摄中断
            logFile.delete();
        }
        java.io.FileOutputStream fos = null;
        try {
            fos = new java.io.FileOutputStream(logFile, true);
            fos.write((line + "\r\n").getBytes("UTF-8"));
        } catch (Throwable ignored) {
            // 写日志失败时静默处理，绝不影响拍摄主流程
        } finally {
            if (fos != null) {
                try {
                    fos.close();
                } catch (Throwable ignored) {
                }
            }
        }
    }
}
``````

```java
package com.pmca.intervalcapture;

/**
 * TimelapseJobWriter —— 可选附加功能：延时视频“合成任务清单”写入器
 * <p>
 * ⚠ 设计原则（严格遵守需求，也是本应用的安全边界）：
 *   1. 照片保存永远是第一优先级。本类只在用户于完成弹窗中**主动选择“是”**时被调用；
 *      默认路径（选择“否”或直接按菜单键）完全不会触达这里；
 *   2. 索尼 PMCA SDK 没有提供“把 SD 卡上已有多张照片编码成一个视频文件”的公开接口，
 *      固件自带的录制接口只能录实时画面。因此机内不会、也无法直接生成延时视频；
 *   3. 本类的做法是：在 SD 卡上生成一份纯文本任务清单 TIMELAPSE_JOB.TXT，
 *      记录本次拍摄的照片目录、张数、建议帧率与间隔，方便之后在电脑上用
 *      ffmpeg 等工具离线合成。全程不转码、不复制、不删除任何原始照片；
 *   4. 任何异常都只返回 false，绝不抛出——视频是附加功能，不能影响照片与拍摄流程。
 * <p>
 * 用到的接口：java.io 文件写入（普通文件 IO），照片本身由相机固件写入，
 * 本类只写一个 KB 级的文本文件，不会对存储卡造成压力。
 */
public final class TimelapseJobWriter {

    private TimelapseJobWriter() {
    }

    /**
     * 写入延时视频合成任务清单。
     *
     * @param takenCount 本次成功保存的照片张数
     * @param settings   本次使用的拍摄参数（用于记录间隔与格式）
     * @return true 表示写入成功
     */
    public static boolean writeJob(int takenCount, CaptureSettings settings) {
        if (takenCount <= 0) {
            // 一张都没拍成，没有可用的素材，直接返回
            return false;
        }
        java.io.FileOutputStream fos = null;
        try {
            // 清单写到 SD 卡 PMCACAP 目录，与日志、参数备份放在一起，方便用户定位
            java.io.File dir = new java.io.File(
                    android.os.Environment.getExternalStorageDirectory().getAbsolutePath(),
                    AppLog.SD_DIR_NAME);
            if (!dir.exists()) {
                dir.mkdirs();
            }
            java.io.File jobFile = new java.io.File(dir, "TIMELAPSE_JOB.TXT");

            // 建议帧率：以“拍摄间隔”换算成 24fps 时间轴需要跳帧还是补帧，
            // 这里只给出一个常见的 24fps 建议值，真正的帧率由用户后期决定。
            int fps = 24;
            float intervalSec = settings.getIntervalSeconds();
            // 每一秒成片所覆盖的真实时长 = fps * 间隔；写在清单里方便后期计算
            int realSecondsPerVideoSecond = (int) (fps * intervalSec);
            // 成片时长（秒）
            float videoSeconds = takenCount / (float) fps;

            StringBuilder sb = new StringBuilder();
            sb.append("PMCA_IntervalCapture 延时视频合成任务清单\r\n");
            sb.append("==========================================\r\n");
            sb.append("本次成功保存照片数：").append(takenCount).append(" 张\r\n");
            sb.append("拍摄间隔：").append(settings.formatInterval()).append("\r\n");
            sb.append("图像格式：").append(settings.formatImageFormat()).append("\r\n");
            sb.append("建议帧率：").append(fps).append(" fps\r\n");
            sb.append("建议成片时长：").append((int) videoSeconds).append(" 秒\r\n");
            sb.append("1 秒成片对应现实时间：").append(realSecondsPerVideoSecond).append(" 秒\r\n");
            sb.append("\r\n");
            sb.append("照片位置：相机存储卡 DCIM 目录下（本应用不移动、不改名照片）\r\n");
            sb.append("\r\n");
            sb.append("电脑端合成示例（把照片按顺序编号后执行，需要 ffmpeg）：\r\n");
            sb.append("  ffmpeg -framerate ").append(fps).append(" -pattern_type glob -i \"*.JPG\" ")
                    .append("-c:v libx264 -pix_fmt yuv420p timelapse.mp4\r\n");
            sb.append("\r\n");
            sb.append("注意：本清单只是一份说明文档，机内不会自动生成任何视频文件。\r\n");
            sb.append("      原始照片请勿删除，它们是唯一的一手素材。\r\n");

            fos = new java.io.FileOutputStream(jobFile, false);
            fos.write(sb.toString().getBytes("UTF-8"));
            fos.flush();
            AppLog.info("已生成延时视频合成清单：" + jobFile.getAbsolutePath());
            return true;
        } catch (Throwable t) {
            AppLog.warn("生成延时视频合成清单失败：" + t.getClass().getSimpleName());
            return false;
        } finally {
            if (fos != null) {
                try {
                    fos.close();
                } catch (Throwable ignored) {
                }
            }
        }
    }
}
``````

## 18. `app\src\main\java\com\pmca\intervalcapture\SystemEventReceiver.java`

**说明**：系统广播接收器（开机完成 / 退出完成，用于日志与状态复位）

```java
package com.pmca.intervalcapture;

import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;

/**
 * SystemEventReceiver —— 接收相机固件的应用生命周期广播
 * <p>
 * 相机固件在下列时机发送广播，应用需要响应，否则会出现“退出后黑屏”
 * 或“再次进入应用时相机被占用”等问题：
 * <pre>
 *   com.android.server.DAConnectionManagerService.BootCompleted
 *       → 固件侧已完成应用侧启动，可以安全操作相机；
 *   com.android.server.DAConnectionManagerService.ExitCompleted
 *       → 固件侧完成了退出流程，本应用需要释放相机资源。
 * </pre>
 * 说明：这是相机固件与 PMCA 应用之间的私有约定（不是安卓官方 API），
 * 名字里的“DAConnectionManager”就是索尼固件里的应用生命周期管理服务。
 * 这里只做日志记录与兜底处理，真正的相机打开/关闭仍由 CaptureActivity 的
 * 生命周期方法负责，避免两处同时操作相机造成冲突。
 */
public class SystemEventReceiver extends BroadcastReceiver {

    /** 固件：应用启动完成 */
    public static final String ACTION_BOOT_COMPLETED =
            "com.android.server.DAConnectionManagerService.BootCompleted";
    /** 固件：应用退出完成 */
    public static final String ACTION_EXIT_COMPLETED =
            "com.android.server.DAConnectionManagerService.ExitCompleted";

    @Override
    public void onReceive(Context context, Intent intent) {
        if (intent == null || intent.getAction() == null) {
            return;
        }
        String action = intent.getAction();
        if (ACTION_BOOT_COMPLETED.equals(action)) {
            // 固件已就绪：记录日志，便于排查相机无法打开的问题
            AppLog.info("收到固件广播：BootCompleted（应用启动完成）");
        } else if (ACTION_EXIT_COMPLETED.equals(action)) {
            // 固件已完成退出：记录日志，正常情况下 Activity 已经释放相机
            AppLog.info("收到固件广播：ExitCompleted（应用退出完成）");
        }
    }
}
``````

```java
package com.pmca.intervalcapture.ui;

import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;
import android.graphics.Rect;

/**
 * UiTheme —— 界面绘制常量与公共绘制方法
 * <p>
 * 相机屏幕逻辑分辨率为 640x480，且本应用交互只用实体按键，因此：
 *   - 坐标全部使用像素绝对值，不做 dpi 换算；
 *   - 所有绘制都走 Canvas + Paint，不使用任何可点击 View；
 *   - 字体统一使用等宽/大字号，保证在相机小屏上可读。
 */
public final class UiTheme {

    /** 逻辑画布宽（与相机屏幕一致） */
    public static final int SCREEN_W = 640;
    /** 逻辑画布高 */
    public static final int SCREEN_H = 480;

    /* ---------------------------- 颜色 ---------------------------- */
    public static final int COLOR_BG = Color.rgb(18, 18, 18);
    public static final int COLOR_PANEL = Color.rgb(32, 32, 34);
    public static final int COLOR_PANEL_FOCUS = Color.rgb(52, 52, 56);
    public static final int COLOR_TITLE_BAR = Color.rgb(0, 96, 128);
    public static final int COLOR_TEXT = Color.rgb(235, 235, 235);
    public static final int COLOR_TEXT_DIM = Color.rgb(150, 150, 150);
    public static final int COLOR_ACCENT = Color.rgb(64, 200, 255);
    public static final int COLOR_WARN = Color.rgb(255, 196, 0);
    public static final int COLOR_ERROR = Color.rgb(255, 72, 72);
    public static final int COLOR_OK = Color.rgb(80, 220, 120);
    public static final int COLOR_REC = Color.rgb(255, 40, 40);

    /* ---------------------------- 字号 ---------------------------- */
    public static final float TEXT_TITLE = 22f;
    public static final float TEXT_LABEL = 18f;
    public static final float TEXT_VALUE = 19f;
    public static final float TEXT_HINT = 14f;
    public static final float TEXT_BIG = 46f;
    public static final float TEXT_HUGE = 72f;

    private UiTheme() {
    }

    /** 创建一个默认 Paint（抗锯齿、无衬线字体） */
    public static Paint newPaint(int color, float size, boolean bold) {
        Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
        p.setColor(color);
        p.setTextSize(size);
        p.setTypeface(bold ? android.graphics.Typeface.DEFAULT_BOLD
                : android.graphics.Typeface.DEFAULT);
        return p;
    }

    /** 画一条水平分隔线 */
    public static void drawHLine(Canvas c, Paint p, int x1, int x2, int y) {
        c.drawLine(x1, y, x2, y, p);
    }

    /** 画矩形边框 */
    public static void drawRectStroke(Canvas c, Paint p, Rect r, float width) {
        p.setStyle(Paint.Style.STROKE);
        p.setStrokeWidth(width);
        c.drawRect(r, p);
        p.setStyle(Paint.Style.FILL);
    }

    /** 在矩形内居中绘制文本（水平+垂直居中） */
    public static void drawTextCentered(Canvas c, Paint p, String text, Rect r) {
        float x = r.exactCenterX();
        float y = r.exactCenterY() - (p.descent() + p.ascent()) / 2f;
        p.setTextAlign(Paint.Align.CENTER);
        c.drawText(text, x, y, p);
        p.setTextAlign(Paint.Align.LEFT);
    }

    /** 左对齐绘制文本，返回文本宽度 */
    public static float drawTextLeft(Canvas c, Paint p, String text, float x, float y) {
        p.setTextAlign(Paint.Align.LEFT);
        c.drawText(text, x, y, p);
        return p.measureText(text);
    }

    /** 右对齐绘制文本 */
    public static void drawTextRight(Canvas c, Paint p, String text, float right, float y) {
        p.setTextAlign(Paint.Align.RIGHT);
        c.drawText(text, right, y, p);
        p.setTextAlign(Paint.Align.LEFT);
    }

    /** 文本宽度 */
    public static float measure(Paint p, String text) {
        return p.measureText(text);
    }

    /** 用等宽数字格式输出“x.x”，避免相机 JVM 上 String.format 的本地化差异 */
    public static String oneDecimal(long tenths) {
        if (tenths < 0) {
            return "0.0";
        }
        long whole = tenths / 10;
        long deci = tenths % 10;
        return whole + "." + deci;
    }

    /** 秒 → “x.x” 文本（用于倒计时） */
    public static String secondsText(long millis) {
        if (millis < 0) {
            return "--.-";
        }
        return oneDecimal(millis / 100L);
    }

    /**
     * 把秒数格式化成“1小时23分45秒”形式，用于预计时长显示。
     */
    public static String durationText(long totalSeconds) {
        if (totalSeconds < 0) {
            return "无限";
        }
        long h = totalSeconds / 3600;
        long m = (totalSeconds % 3600) / 60;
        long s = totalSeconds % 60;
        StringBuilder sb = new StringBuilder();
        if (h > 0) {
            sb.append(h).append("小时");
        }
        if (m > 0) {
            sb.append(m).append("分");
        }
        sb.append(s).append("秒");
        return sb.toString();
    }
}
``````

```java
package com.pmca.intervalcapture.ui;

import android.graphics.Canvas;
import android.graphics.Paint;
import android.graphics.Rect;

import com.pmca.intervalcapture.CaptureSettings;
import com.pmca.intervalcapture.IntervalScheduler;

/**
 * RunScreen —— 运行状态界面（拍摄中 / 暂停 / 已结束 / 异常）
 * <p>
 * 640x480 布局：
 * <pre>
 *  y  0~40   标题栏：状态文字（待机/拍摄中/已暂停/已结束/异常停止） + 录制红点
 *  y 40~190  “已拍张数”大号数字 + 剩余张数 + 进度条
 *  y190~300  “距下一张 x.x 秒”大号倒计时 + 最近一张耗时
 *  y300~352  参数摘要（间隔 / 张数 / 格式 / 锁定 / 预览）
 *  y352~400  健康状态（机身温度等级 / 存储卡剩余 / 异常说明）
 *  y400~440  提示信息（最近一次警告或错误）
 *  y440~480  按键提示
 * </pre>
 * 画面右侧预留一块区域给相机预览（由 Activity 的 SurfaceView 覆盖），
 * 保证“预览开”时能实时看画面、“预览关”时这块区域就是纯黑。
 */
public final class RunScreen {

    /** 预览区域（逻辑坐标），SurfaceView 会按这个矩形对齐 */
    public static final Rect PREVIEW_RECT = new Rect(344, 40, 640, 340);

    private final Paint titlePaint = UiTheme.newPaint(UiTheme.COLOR_TEXT, UiTheme.TEXT_TITLE, true);
    private final Paint bigPaint = UiTheme.newPaint(UiTheme.COLOR_TEXT, UiTheme.TEXT_BIG, true);
    private final Paint hugePaint = UiTheme.newPaint(UiTheme.COLOR_ACCENT, UiTheme.TEXT_HUGE, true);
    private final Paint labelPaint = UiTheme.newPaint(UiTheme.COLOR_TEXT_DIM, UiTheme.TEXT_LABEL, false);
    private final Paint textPaint = UiTheme.newPaint(UiTheme.COLOR_TEXT, UiTheme.TEXT_VALUE, false);
    private final Paint hintPaint = UiTheme.newPaint(UiTheme.COLOR_TEXT_DIM, UiTheme.TEXT_HINT, false);
    private final Paint warnPaint = UiTheme.newPaint(UiTheme.COLOR_WARN, UiTheme.TEXT_HINT, true);
    private final Paint errPaint = UiTheme.newPaint(UiTheme.COLOR_ERROR, UiTheme.TEXT_HINT, true);
    private final Paint okPaint = UiTheme.newPaint(UiTheme.COLOR_OK, UiTheme.TEXT_HINT, true);
    private final Paint panelPaint = UiTheme.newPaint(UiTheme.COLOR_PANEL, 1f, false);
    private final Paint barBgPaint = UiTheme.newPaint(UiTheme.COLOR_PANEL_FOCUS, 1f, false);
    private final Paint barFgPaint = UiTheme.newPaint(UiTheme.COLOR_OK, 1f, false);
    private final Paint recPaint = UiTheme.newPaint(UiTheme.COLOR_REC, 1f, false);
    private final Paint barPaint = UiTheme.newPaint(UiTheme.COLOR_TITLE_BAR, 1f, false);

    /** 状态栏颜色随状态变化 */
    private int statusColor(int state) {
        switch (state) {
            case IntervalScheduler.STATE_RUNNING:
                return UiTheme.COLOR_REC;
            case IntervalScheduler.STATE_PAUSED:
                return UiTheme.COLOR_WARN;
            case IntervalScheduler.STATE_ERROR:
                return UiTheme.COLOR_ERROR;
            case IntervalScheduler.STATE_FINISHED:
                return UiTheme.COLOR_OK;
            default:
                return UiTheme.COLOR_ACCENT;
        }
    }

    /**
     * 绘制运行界面。
     *
     * @param canvas      画布
     * @param scheduler   调度器（读取计数、倒计时、状态）
     * @param settings    本次任务使用的参数快照
     * @param tempText    机身温度描述
     * @param freeText    存储卡剩余空间描述
     * @param message     最近一次提示（成功/警告/错误），可为 null
     * @param messageIsError 提示是否表示错误
     * @param showPreview 是否显示预览区域（关闭时该区域画成纯黑并写“预览已关闭”）
     */
    public void draw(Canvas canvas, IntervalScheduler scheduler, CaptureSettings settings,
                     String tempText, String freeText, String message, boolean messageIsError,
                     boolean showPreview) {
        canvas.drawColor(UiTheme.COLOR_BG);
        int state = scheduler.getState();

        // ---------- 标题栏 ----------
        Rect titleBar = new Rect(0, 0, UiTheme.SCREEN_W, 40);
        canvas.drawRect(titleBar, barPaint);
        // 拍摄中时在标题左侧画一个红色圆点（模拟录制指示）
        int titleX = 16;
        if (state == IntervalScheduler.STATE_RUNNING) {
            canvas.drawCircle(26, 20, 8, recPaint);
            titleX = 44;
        }
        canvas.drawText("间隔拍摄  |  " + IntervalScheduler.stateName(state), titleX, 28, titlePaint);
        hintPaint.setTextAlign(Paint.Align.RIGHT);
        canvas.drawText("间隔 " + settings.formatInterval(), UiTheme.SCREEN_W - 16, 27, hintPaint);
        hintPaint.setTextAlign(Paint.Align.LEFT);

        // ---------- 左侧信息区（宽 344）----------
        int leftW = PREVIEW_RECT.left;

        // 已拍张数（大号）
        canvas.drawText("已拍张数", 16, 78, labelPaint);
        int taken = scheduler.getTakenCount();
        // 无限模式张数可能很大，字号自适应
        hugePaint.setTextSize(taken >= 10000 ? 54f : 72f);
        UiTheme.drawTextRight(canvas, hugePaint, String.valueOf(taken), leftW - 16, 150);

        // 剩余张数 / 总张数
        int remain = scheduler.getRemainingCount();
        String totalText = settings.isInfiniteCount()
                ? "总张数：无限"
                : "总张数：" + settings.getEffectiveShotCount();
        canvas.drawText("剩余张数：" + (remain < 0 ? "无限" : String.valueOf(remain)), 16, 186, textPaint);
        canvas.drawText(totalText, 16, 208, textPaint);

        // 进度条（固定张数模式）
        int barLeft = 16, barRight = leftW - 16, barY = 218;
        canvas.drawRect(new Rect(barLeft, barY, barRight, barY + 10), barBgPaint);
        int percent = scheduler.getProgressPercent();
        if (percent < 0) {
            // 无限模式：用“已拍张数对 100 的余数”做一个循环进度，纯装饰
            int cyc = taken % 100;
            int w = (barRight - barLeft) * cyc / 100;
            canvas.drawRect(new Rect(barLeft, barY, barLeft + w, barY + 10), barFgPaint);
        } else {
            int w = (barRight - barLeft) * percent / 100;
            canvas.drawRect(new Rect(barLeft, barY, barLeft + w, barY + 10), barFgPaint);
            UiTheme.drawTextRight(canvas, hintPaint, percent + "%", barRight, barY + 28);
        }

        // ---------- 倒计时 ----------
        canvas.drawText("距下一张", 16, 272, labelPaint);
        long countdown = scheduler.getCountdownMillis();
        String cdText;
        if (state == IntervalScheduler.STATE_PAUSED) {
            cdText = "--.-";
        } else if (state == IntervalScheduler.STATE_FINISHED || state == IntervalScheduler.STATE_ERROR) {
            cdText = "--.-";
        } else if (scheduler.isShotInFlight()) {
            cdText = "拍摄中";
            hugePaint.setTextSize(48f);
        } else {
            cdText = UiTheme.secondsText(countdown) + " 秒";
            hugePaint.setTextSize(64f);
        }
        UiTheme.drawTextRight(canvas, hugePaint, cdText, leftW - 16, 324);
        hugePaint.setTextSize(72f);

        // 最近一张耗时 / 失败张数 / 节拍溢出次数
        long lastMs = scheduler.getLastShotDurationMs();
        canvas.drawText("上张耗时 " + UiTheme.secondsText(lastMs) + " 秒", 16, 352, hintPaint);
        String fails = "失败 " + scheduler.getFailedCount() + " 张";
        if (scheduler.getOverrunCount() > 0) {
            fails += "   节拍溢出 " + scheduler.getOverrunCount() + " 次";
        }
        canvas.drawText(fails, 16, 372, scheduler.getOverrunCount() > 0 ? warnPaint : hintPaint);

        // ---------- 健康状态 ----------
        canvas.drawText("机身温度：" + tempText, 16, 398, hintPaint);
        canvas.drawText("存储卡：" + freeText, 16, 418, hintPaint);

        // ---------- 提示区 ----------
        if (message != null) {
            Paint mp = messageIsError ? errPaint : warnPaint;
            canvas.drawText(truncate(mp, message, UiTheme.SCREEN_W - 32), 16, 442, mp);
        }

        // ---------- 预览区域 ----------
        Rect pr = PREVIEW_RECT;
        canvas.drawRect(pr, panelPaint);
        if (!showPreview) {
            Paint p = UiTheme.newPaint(UiTheme.COLOR_TEXT_DIM, UiTheme.TEXT_HINT, false);
``````

## 23. `tools\build-apk.ps1`

**说明**：一键打包脚本（aapt → javac → d8 → 合并 → 对齐 → 签名 → 校验）

```powershell
# 一键构建 PMCA_IntervalCapture 的 APK（Windows PowerShell 版）
#
# 流程：aapt 打包资源+清单 → javac 编译 → d8 生成 classes.dex → 组装 zip
#       → zipalign 对齐 → apksigner 签名
#
# 用法（在仓库根目录执行）：
#   powershell -ExecutionPolicy Bypass -File tools\build-apk.ps1
#
# 前置要求见 docs/01-编译环境搭建.md：JDK 8+、Android build-tools、android.jar
param(
    [string]$BuildTools = "$PSScriptRoot\..\..\build-support\android-tools\android-15",
    [string]$AndroidJar = "$PSScriptRoot\..\..\build-support\libs\android-4.1.1.4.jar",
    [string]$SonyStubJar = "$PSScriptRoot\..\..\build-support\libs\sony-pmca-stubs.jar",
    [string]$KeyStore = "$PSScriptRoot\..\build-support\pmca-debug.keystore",
    [string]$OutDir = "$PSScriptRoot\..\build\outputs"
)

$ErrorActionPreference = 'Stop'
$root = Resolve-Path "$PSScriptRoot\.."
$srcDir = Join-Path $root 'app\src\main'
$work = Join-Path $root 'build\work'
$resDir = Join-Path $srcDir 'res'
$manifest = Join-Path $srcDir 'AndroidManifest.xml'

Write-Host "== 1/8 清理工作目录 ==" -ForegroundColor Cyan
if (Test-Path $work) { Remove-Item -Recurse -Force $work }
New-Item -ItemType Directory -Force -Path $work, (Join-Path $work 'gen'), (Join-Path $work 'classes'), (Join-Path $work 'dex'), $OutDir | Out-Null

Write-Host "== 2/8 生成 R.java 并打包资源 ==" -ForegroundColor Cyan
# aapt package：把 res/ 编译成 resources.arsc 并生成 R.java（平台资源 id 使用 android.jar 中的定义）
& "$BuildTools\aapt.exe" package -f -m -J (Join-Path $work 'gen') -M $manifest -S $resDir -I $AndroidJar --auto-add-overlay
if ($LASTEXITCODE -ne 0) { throw "aapt package 失败" }

# 生成未签名的 APK 骨架（含清单、资源、resources.arsc）
& "$BuildTools\aapt.exe" package -f -M $manifest -S $resDir -I $AndroidJar -F (Join-Path $work 'base.apk')
if ($LASTEXITCODE -ne 0) { throw "aapt 生成 base.apk 失败" }

Write-Host "== 3/8 编译 Java 源码 ==" -ForegroundColor Cyan
$javaFiles = Get-ChildItem -Recurse -Path @((Join-Path $srcDir 'java'), (Join-Path $work 'gen')) -Filter *.java |
    ForEach-Object { $_.FullName }
$cp = "$AndroidJar;$SonyStubJar"
& javac -nowarn -encoding UTF-8 -source 8 -target 8 -cp $cp -d (Join-Path $work 'classes') $javaFiles
if ($LASTEXITCODE -ne 0) { throw "javac 编译失败" }

Write-Host "== 4/8 使用 d8 生成 classes.dex ==" -ForegroundColor Cyan
$classFiles = Get-ChildItem -Recurse -Path (Join-Path $work 'classes') -Filter *.class | ForEach-Object { $_.FullName }
# --min-api 10：生成 Android 2.3（相机运行时）可用的 DEX；
# --lib：把 android.jar 声明为“平台类库”，d8 不会把它打进 dex
& cmd /c "`"$BuildTools\d8.bat`" --min-api 10 --release --lib `"$AndroidJar`" --lib `"$SonyStubJar`" --output `"$(Join-Path $work 'dex')`" $($classFiles -join ' ')"
if ($LASTEXITCODE -ne 0) { throw "d8 生成 DEX 失败" }

Write-Host "== 5/8 组装 APK ==" -ForegroundColor Cyan
$unaligned = Join-Path $work 'app-unaligned.apk'
Copy-Item (Join-Path $work 'base.apk') $unaligned -Force
# 把 classes.dex 加进 APK 根目录（Android 运行时从这里加载）
Add-Type -AssemblyName System.IO.Compression.FileSystem
$zip = [System.IO.Compression.ZipFile]::Open($unaligned, 'Update')
$dexEntry = $zip.CreateEntry('classes.dex', [System.IO.Compression.CompressionLevel]::Optimal)
$dexStream = $dexEntry.Open()
$dexBytes = [System.IO.File]::ReadAllBytes((Join-Path $work 'dex\classes.dex'))
$dexStream.Write($dexBytes, 0, $dexBytes.Length)
$dexStream.Close()
$zip.Dispose()

Write-Host "== 6/8 zipalign 对齐 ==" -ForegroundColor Cyan
$aligned = Join-Path $work 'app-aligned.apk'
& "$BuildTools\zipalign.exe" -f -p 4 $unaligned $aligned
if ($LASTEXITCODE -ne 0) { throw "zipalign 失败" }

Write-Host "== 7/8 签名 ==" -ForegroundColor Cyan
if (-not (Test-Path (Split-Path $KeyStore))) { New-Item -ItemType Directory -Force -Path (Split-Path $KeyStore) | Out-Null }
if (-not (Test-Path $KeyStore)) {
    Write-Host "   未找到密钥库，自动生成自签名密钥库（仅用于 PMCA-RE 刷入测试）"
        & keytool -genkeypair -v -keystore $KeyStore -storepass pmca123456 -keypass pmca123456 -alias pmca -keyalg RSA -keysize 2048 -validity 10000 -dname "CN=PMCA IntervalCapture, OU=DIY, O=PMCA, L=Beijing, ST=Beijing, C=CN" 2>&1 | Out-Null
    if ($LASTEXITCODE -ne 0) { throw "生成密钥库失败" }
}
$final = Join-Path $OutDir 'PMCA_IntervalCapture.apk'
# 必须使用 V1（JAR）签名：相机运行时是 Android 2.3，只认 V1 签名。
# 这里显式关闭 v2/v3/v3.1/v4，保证产物只带一个纯 JAR 签名块——最贴近相机固件的校验方式，
# 也便于用“解包看 META-INF/*.SF/*.RSA”这种最朴素的方式自查。
& cmd /c "`"$BuildTools\apksigner.bat`" sign --ks `"$KeyStore`" --ks-pass pass:pmca123456 --key-pass pass:pmca123456 --ks-key-alias pmca --v1-signing-enabled true --v2-signing-enabled false --v3-signing-enabled false --v4-signing-enabled false --min-sdk-version 10 --out `"$final`" `"$aligned`""
if ($LASTEXITCODE -ne 0) { throw "apksigner 签名失败" }

Write-Host "== 8/8 校验 ==" -ForegroundColor Cyan
& cmd /c "`"$BuildTools\apksigner.bat`" verify --verbose `"$final`""
& cmd /c "`"$BuildTools\aapt.exe`" dump badging `"$final`"" | Select-Object -First 6

Write-Host ""
Write-Host "构建完成：$final" -ForegroundColor Green
Write-Host "APK 大小：$([math]::Round((Get-Item $final).Length / 1KB, 1)) KB"
``````

```java
import com.pmca.intervalcapture.CaptureSettings;

public final class LogicTest {

    private static int passed = 0;
    private static int failed = 0;

    public static void main(String[] args) throws Exception {
        testIntervalClamp();
        testIntervalStep();
        testShotCountClamp();
        testFormatCycle();
        testPersistRoundTrip();

        testSchedulerIntervalTiming();

            System.exit(1);
        }
    }

    /* ============================ 断言工具 ============================ */

    private static void check(String name, boolean cond, String detail) {
        if (cond) {
            passed++;
            System.out.println("[PASS] " + name);
        } else {
            failed++;
            System.out.println("[FAIL] " + name + "  -> " + detail);
        }
    }

    private static void eq(String name, long expect, long actual) {
        check(name, expect == actual, "期望 " + expect + "，实际 " + actual);
    }

    private static void eq(String name, String expect, String actual) {
        check(name, expect == null ? actual == null : expect.equals(actual),
                "期望 " + expect + "，实际 " + actual);
    }

    /* ============================ 参数模型测试 ============================ */

    private static void testIntervalClamp() {
        CaptureSettings s = new CaptureSettings();
        s.setIntervalDeciseconds(1);          // 低于下限 1.0 秒
        eq("间隔下限钳制为 1.0 秒", 10, s.getIntervalDeciseconds());
        s.setIntervalDeciseconds(999999);     // 超过上限 3600 秒
        eq("间隔上限钳制为 3600 秒", 36000, s.getIntervalDeciseconds());
        s.setIntervalDeciseconds(25);
        eq("间隔 2.5 秒可正常设置", 25, s.getIntervalDeciseconds());
        eq("间隔 2.5 秒格式化为文本", "2.5 秒", s.formatInterval());
        s.setIntervalDeciseconds(20);
        eq("整秒格式化不带小数点", "2 秒", s.formatInterval());
    }

    private static void testIntervalStep() {
        CaptureSettings s = new CaptureSettings();
        s.setIntervalDeciseconds(10);        // 1.0 秒

        eq("1.0 秒向上一步为 1.1 秒", 11, s.getIntervalDeciseconds());
        s.adjust(CaptureSettings.FIELD_INTERVAL, -1);
        eq("1.1 秒向下一步回到 1.0 秒", 10, s.getIntervalDeciseconds());
        s.adjust(CaptureSettings.FIELD_INTERVAL, -1);
``````
